    // The direct internal admission seam remains available for existing focused qualification.
    internal CoreLogRuntime(EventMetadata metadata, LoggingOptions options, ILogOutputAdmission? console,
        ILogOutputAdmission? file, TimeProvider? timeProvider = null, object? sharedAdmissionGate = null, OutputQueues? queuedOutputs = null, ConsoleOutputWriter? consoleWriter = null, FileOutputWriter? fileWriter = null)
    {
        ArgumentNullException.ThrowIfNull(metadata);
        ArgumentNullException.ThrowIfNull(options);
        CoreLogLevel consoleLevel = options.EffectiveConsoleLevel;
        CoreLogLevel fileLevel = options.EffectiveFileLevel;
        bool consoleEnabled = consoleLevel != CoreLogLevel.None;
        bool fileEnabled = !string.IsNullOrWhiteSpace(options.FilePath) && fileLevel != CoreLogLevel.None;
        int maximumBytes = options.MaxEntryBytes;
        ValidateSettings(consoleLevel, fileLevel, maximumBytes);
        ValidateShutdownTimeout(options.ShutdownTimeoutMilliseconds);
        admissionGate = sharedAdmissionGate ?? new object();
        this.queuedOutputs = queuedOutputs;
        this.consoleWriter = consoleWriter;
        this.fileWriter = fileWriter;
        recoveredPartialSegment = fileWriter?.RecoveredPartialSegment == true;
        this.metadata = metadata;
        this.timeProvider = timeProvider ?? TimeProvider.System;
        maximumEntryBytes = maximumBytes;
        shutdownTimeout = options.ShutdownTimeout;
        var configured = new List<Output>(2);
        Add("console", consoleEnabled, consoleLevel, console);
        Add("file", fileEnabled, fileLevel, file);
        outputs = configured.ToArray();

        void Add(string name, bool enabled, CoreLogLevel threshold, ILogOutputAdmission? admission)
        {
            if (enabled != (admission is not null))
            {
                throw new ArgumentException("Output composition must match enabled settings.", name);
            }
            if (admission is not null) { configured.Add(new Output(name, threshold, admission)); }
        }
    }

    /// <summary>Creates an application-owned runtime using the prepared static environment settings and automatically captured facts.</summary>
    /// <remarks>Configure EnvironmentOptions.Logging before creation; preparation normalizes its FilePath in place. Concurrent mutation of those settings is unsupported. Each runtime owns only its own workers and file group. DisposeAsync joins its bounded shutdown.</remarks>
    /// <exception cref="CIS.Configuration.CoreConfigurationException">Module settings or captured metadata are invalid.</exception>
    /// <exception cref="TypeInitializationException">Static environment initialization failed.</exception>
    /// <exception cref="CoreLogInitializationException">An enabled output cannot initialize.</exception>
    public static CoreLogRuntime Create()
    {
        LoggingOptions options = EnvironmentOptions.Logging;
        options.Prepare(Directory.GetCurrentDirectory());
        return CreateConfiguredQueued(new EventMetadata(), options);
    }

    // Actual capture -> ticketed Channels -> sequential trusted writers; physical output ownership comes later.
    internal static CoreLogRuntime CreateQueued(EventMetadata metadata, LoggingOptions options,
        Func<AdmittedEvent, ValueTask>? consoleWrite, Func<AdmittedEvent, ValueTask>? fileWrite, TimeProvider? timeProvider = null,
        Func<ValueTask>? consoleFlush = null, Func<ValueTask>? fileFlush = null)
    {
        ArgumentNullException.ThrowIfNull(metadata);
        return ComposeQueued(metadata, CaptureQueueOptions(options), consoleWrite, fileWrite, timeProvider, consoleFlush: consoleFlush, fileFlush: fileFlush);
    }

    // Console settings are frozen before selecting the stream or starting any output worker.
    internal static CoreLogRuntime CreateConsoleQueued(EventMetadata metadata, LoggingOptions options,
        Func<AdmittedEvent, ValueTask>? fileWrite = null, TimeProvider? timeProvider = null,
        Func<ConsoleOutputStream, Stream>? acquireConsole = null, Func<Func<Task>, Task>? launchWorker = null)
    {
        ArgumentNullException.ThrowIfNull(metadata);
        LoggingOptions selected = CaptureQueueOptions(options);
        ValidateSettings(selected.EffectiveConsoleLevel, selected.EffectiveFileLevel, selected.MaxEntryBytes);
        if (selected.QueueCapacity is < LoggingOptions.MinimumQueueCapacity or > LoggingOptions.MaximumQueueCapacity)
        { throw new ArgumentOutOfRangeException(nameof(options), "Use the configured queue-capacity range."); }
        if (!Enum.IsDefined(selected.ConsoleFormat) || !Enum.IsDefined(selected.HumanDetail) || !Enum.IsDefined(selected.ConsoleStream))
        { throw new ArgumentException("Use declared console presentation and stream values.", nameof(options)); }
        ConsoleOutputWriter? writer = AcquireConsoleWriter(selected, acquireConsole);
        return ComposeQueued(metadata, selected, writer is null ? null : writer.WriteAsync, fileWrite, timeProvider, writer, launchWorker);
    }

    internal static CoreLogRuntime CreateConfiguredQueued(EventMetadata metadata, LoggingOptions options, TimeProvider? timeProvider = null,
        Func<ConsoleOutputStream, Stream>? acquireConsole = null, Func<Func<Task>, Task>? launchWorker = null)
    {
        ArgumentNullException.ThrowIfNull(metadata);
        LoggingOptions selected = CaptureQueueOptions(options);
        ValidateSettings(selected.EffectiveConsoleLevel, selected.EffectiveFileLevel, selected.MaxEntryBytes);
        if (selected.QueueCapacity is < LoggingOptions.MinimumQueueCapacity or > LoggingOptions.MaximumQueueCapacity
            || selected.FileMaxBytes is < LoggingOptions.MinimumFileMaxBytes or > LoggingOptions.MaximumFileMaxBytes
            || selected.FileRetainedFiles is < LoggingOptions.MinimumFileRetainedFiles or > LoggingOptions.MaximumFileRetainedFiles
            || selected.FileEnabled && (selected.FileMaxBytes < selected.MaxEntryBytes
                || selected.FlushIntervalMilliseconds is < LoggingOptions.MinimumFlushIntervalMilliseconds or > LoggingOptions.MaximumFlushIntervalMilliseconds))
        { throw new ArgumentOutOfRangeException(nameof(options)); }
        if (!Enum.IsDefined(selected.ConsoleFormat) || !Enum.IsDefined(selected.ConsoleStream) || !Enum.IsDefined(selected.FileFormat) || !Enum.IsDefined(selected.HumanDetail))
        { throw new ArgumentException("Use declared output presentation values.", nameof(options)); }
        if (selected.FileEnabled && !Path.IsPathFullyQualified(selected.FilePath!))
        { throw new ArgumentException("File output requires an already captured absolute path.", nameof(options)); }
        ConsoleOutputWriter? console = AcquireConsoleWriter(selected, acquireConsole);
        FileOutputWriter? file = selected.FileEnabled ? new(selected.FilePath!, selected.FileFormat, selected.HumanDetail, selected.MaxEntryBytes, selected.FileMaxBytes, selected.FileRetainedFiles) : null;
        return ComposeQueued(metadata, selected, console is null ? null : console.WriteAsync, file is null ? null : file.WriteAsync, timeProvider, console, launchWorker, file);
    }

    private static ConsoleOutputWriter? AcquireConsoleWriter(LoggingOptions selected, Func<ConsoleOutputStream, Stream>? acquireConsole)
    {
        ConsoleOutputWriter? writer = null;
        if (selected.ConsoleEnabled)
        {
            try
            {
                Stream stream = acquireConsole is null ?
                    selected.ConsoleStream == ConsoleOutputStream.StandardError ? Console.OpenStandardError() : Console.OpenStandardOutput() :
                    acquireConsole(selected.ConsoleStream);
                if (stream is null) { throw new CoreLogInitializationException("console", "ConsoleInitializationFailed"); }
                writer = new ConsoleOutputWriter(stream, selected.ConsoleFormat, selected.HumanDetail, selected.MaxEntryBytes);
            }
            catch (CoreLogInitializationException) { throw; }
            catch { throw new CoreLogInitializationException("console", "ConsoleInitializationFailed"); }
        }
        return writer;
    }

    private static LoggingOptions CaptureQueueOptions(LoggingOptions options)
    {
        ArgumentNullException.ThrowIfNull(options);
        ValidateShutdownTimeout(options.ShutdownTimeoutMilliseconds);
        return new LoggingOptions
        {
            ConsoleLevel = options.EffectiveConsoleLevel,
            FileLevel = options.EffectiveFileLevel,
            FilePath = options.FilePath,
            MaxEntryBytes = options.MaxEntryBytes,
            QueueCapacity = options.QueueCapacity,
            ConsoleFormat = options.ConsoleFormat,
            ConsoleStream = options.ConsoleStream,
            HumanDetail = options.HumanDetail,
            FileFormat = options.FileFormat,
            FileMaxBytes = options.FileMaxBytes,
            FileRetainedFiles = options.FileRetainedFiles,
            FlushIntervalMilliseconds = options.FlushIntervalMilliseconds,
            ShutdownTimeoutMilliseconds = options.ShutdownTimeoutMilliseconds
        };
    }

    private static CoreLogRuntime ComposeQueued(EventMetadata metadata, LoggingOptions selected,
        Func<AdmittedEvent, ValueTask>? consoleWrite, Func<AdmittedEvent, ValueTask>? fileWrite,
        TimeProvider? timeProvider, ConsoleOutputWriter? writer = null, Func<Func<Task>, Task>? launchWorker = null, FileOutputWriter? fileWriter = null, Func<ValueTask>? consoleFlush = null, Func<ValueTask>? fileFlush = null)
    {
        ValidateSettings(selected.EffectiveConsoleLevel, selected.EffectiveFileLevel, selected.MaxEntryBytes);
        var gate = new object();
        var queues = new OutputQueues(selected.QueueCapacity, selected.ConsoleEnabled, selected.FileEnabled, consoleWrite, fileWrite, gate, launchWorker, fileWriter,
            writer is null ? consoleFlush : writer.FlushAsync, fileWriter is null ? fileFlush : fileWriter.FlushAsync,
            writer is null ? null : writer.WriteAsync, fileWriter is null ? null : fileWriter.WriteAsync,
            writer is null ? null : writer.FlushAsync, fileWriter is null ? null : fileWriter.FlushAsync);
        try
        {
            writer?.Initialize();
            fileWriter?.Initialize();
            var runtime = new CoreLogRuntime(metadata, selected, queues.Console, queues.File, timeProvider, gate, queues, writer, fileWriter);
            if (fileWriter is not null)
            {
                try { queues.File!.StartFileFlushTimer(runtime.timeProvider, selected.FlushInterval); }
                catch { throw new CoreLogInitializationException("file", "FileFlushTimerInitializationFailed"); }
            }
            return runtime;
        }
        catch
        {
            queues.Complete();
            // Observe terminal cleanup failures without masking the safe initialization category.
            try { queues.Completion.GetAwaiter().GetResult(); } catch { }
            try { fileWriter?.Dispose(); } catch { }
            throw;
        }
    }
    internal OutputQueues? QueuedOutputs => queuedOutputs;
    private static void ValidateShutdownTimeout(int milliseconds)
    {
        if (milliseconds is < LoggingOptions.MinimumShutdownTimeoutMilliseconds or > LoggingOptions.MaximumShutdownTimeoutMilliseconds)
        { throw new ArgumentOutOfRangeException(nameof(milliseconds), "Use the configured shutdown-timeout range."); }
    }

    /// <summary>Starts or joins one shared bounded stop operation through its original accepted boundaries.</summary>
    /// <param name="cancellationToken">Cancels only this caller's wait; even an already-cancelled first caller starts shutdown.</param>
    /// <returns>The same owned immutable observation retained for all joining callers.</returns>
    /// <exception cref="OperationCanceledException">The caller cancels its own wait.</exception>
    /// <remarks>The configured overall timeout is frozen at creation. UTC is an observation label; elapsed time is monotonic. At expiry, unstarted accepted records are abandoned. Cancellation of active writes/flushes is requested but is not guaranteed, and no new flush, retry or replacement writer starts. Completed means terminal dispositions, not successful delivery or resource release. A timed-out observation can retain pending I/O; its later completion does not change the cached result.</remarks>
    public Task<LogStopResult> StopAsync(CancellationToken cancellationToken)
    {
        if (queuedOutputs is null) { throw new InvalidOperationException("Stop requires actual queued output composition."); }
        Task<LogStopResult>? shared;
        lock (admissionGate) { shared = stopTask; }
        if (shared is null)
        {
            // Caller clock code is never invoked under admission synchronization.
            // Preliminary concurrent samples are harmless; only the winning caller owns the deadline.
            long started = timeProvider.GetTimestamp();
            RuntimeFlushBoundary boundary = default;
            TaskCompletionSource<LogStopResult>? completion = null;
            lock (admissionGate)
            {
                if (stopTask is null)
                {
                    completion = new(TaskCreationOptions.RunContinuationsAsynchronously);
                    stopTask = completion.Task;
                    try
                    {
                        // Register final controls before a closing idle worker is allowed to exit.
                        boundary = RegisterFlushBoundary();
                        shutdownStartedAtUtc ??= boundary.BoundaryCapturedAtUtc;
                        admissionClosed = true;
                        queuedOutputs.Complete();
                    }
                    catch (Exception error)
                    {
                        // A failed setup must not leave the already-published task pending forever.
                        completion.TrySetException(error);
                        throw;
                    }
                }
                shared = stopTask;
            }
            if (completion is not null)
            {
                // Start deadline acquisition directly, without a delayed ThreadPool launch.
                // Only continuations/timers outlive this call; they do not retain the caller's flow.
                if (ExecutionContext.IsFlowSuppressed()) { _ = RunStopAsync(boundary, started, completion); }
                else { using (ExecutionContext.SuppressFlow()) { _ = RunStopAsync(boundary, started, completion); } }
            }
        }
        // WaitAsync can prefer an already-completed task; an already-cancelled caller must still cancel.
        if (cancellationToken.IsCancellationRequested) { return Task.FromCanceled<LogStopResult>(cancellationToken); }
        return cancellationToken.CanBeCanceled ? shared!.WaitAsync(cancellationToken) : shared!;
    }

    /// <summary>Starts or joins the original bounded stop operation without a new drain window.</summary>
    /// <returns>A wait on the same shared operation; unsuccessful draining remains visible in LastStopResult.</returns>
    /// <remarks>Repeated disposal is harmless. At the original deadline this wait may complete while actual I/O and owned cleanup remain pending; no resource is disposed beneath active I/O. Inspect GetDiagnostics().LastStopResult for the immutable drain outcome.</remarks>
    public ValueTask DisposeAsync() => new(StopAsync(CancellationToken.None));

    private async Task RunStopAsync(RuntimeFlushBoundary boundary, long started, TaskCompletionSource<LogStopResult> completion)
    {
        try
        {
            if (boundary.Console is null && boundary.File is null)
            { PublishStop(CaptureFlushResult(boundary), completion); return; }
            await using var deadline = new FlushDeadline(timeProvider, started, shutdownTimeout, () => CaptureFlushResult(boundary), () => CaptureStopDeadline(boundary));
            try
            {
                await boundary.WaitForWriterFlushAsync(deadline.Token).ConfigureAwait(false);
                if (boundary.Console is { } console)
                { await queuedOutputs!.Console!.WaitForCoveredRecordsAsync("console", console, deadline.Token).ConfigureAwait(false); }
                if (boundary.File is { } file)
                { await queuedOutputs!.File!.WaitForCoveredRecordsAsync("file", file, deadline.Token).ConfigureAwait(false); }
                try { await queuedOutputs!.Completion.WaitAsync(deadline.Token).ConfigureAwait(false); }
                catch (Exception) when ((queuedOutputs!.Completion.IsFaulted || queuedOutputs.Completion.IsCanceled) && (queuedOutputs.Console?.IsFaulted == true || queuedOutputs.File?.IsFaulted == true))
                { /* Terminal writer faults remain safe output results, not business-call I/O exceptions. */ }
            }
            catch (OperationCanceledException) when (deadline.Token.IsCancellationRequested) { }
            // Publish the frozen result before asynchronous deadline-timer cleanup can wait on a callback join.
            PublishStop(deadline.ObserveCompletion(), completion);
        }
        catch (Exception error)
        {
            if (completion.Task.IsCompletedSuccessfully)
            {
                // Publication precedes deadline-timer cleanup; later failure belongs to live diagnostics.
                queuedOutputs!.RecordCleanupFault("ShutdownTimerCleanupFailed");
            }
            else { completion.TrySetException(error); }
        }
    }

    private LogFlushResult CaptureStopDeadline(RuntimeFlushBoundary boundary)
    {
        LogFlushResult observation;
        lock (admissionGate)
        {
            queuedOutputs!.AbandonUnstarted();
            observation = CaptureFlushResult(boundary);
        }
        // Freeze actual pending work before cooperative cancellation can settle it.
        queuedOutputs.RequestIoCancellation();
        return observation;
    }
    private void PublishStop(LogFlushResult observation, TaskCompletionSource<LogStopResult> completion)
    {
        var result = new LogStopResult(observation);
        lock (admissionGate) { lastStopResult = result; }
        completion.TrySetResult(result);
    }


    /// <summary>Waits through fixed accepted boundaries and confirms each enabled writer's buffer flush.</summary>
    /// <param name="timeout">One positive finite overall wait budget; every positive TimeSpan is supported.</param>
    /// <param name="cancellationToken">Cancels only this caller's wait, preserving shared worker requests.</param>
    /// <returns>An owned immutable observation, including unresolved work at the deadline.</returns>
    /// <exception cref="ArgumentOutOfRangeException">The timeout is zero, negative or infinite.</exception>
    /// <exception cref="OperationCanceledException">The caller token is cancelled, including after boundary registration.</exception>
    /// <remarks>UTC labels are system observations; elapsed deadlines use the runtime TimeProvider. Flush does not stop admission or guarantee disk durability. After closure it cannot reopen a writer; a terminal writer returns an unavailable/failed flush even if the original stop flushed successfully. Inspect LastStopResult for that original shutdown outcome.</remarks>
    public Task<LogFlushResult> FlushAsync(TimeSpan timeout, CancellationToken cancellationToken)
    {
        if (timeout.Ticks <= 0) { throw new ArgumentOutOfRangeException(nameof(timeout)); }
        long started = timeProvider.GetTimestamp();
        RuntimeFlushBoundary boundary = RegisterFlushBoundary();
        return ObserveFlushAsync(boundary, started, timeout, cancellationToken);
    }
    private async Task<LogFlushResult> ObserveFlushAsync(RuntimeFlushBoundary boundary, long started, TimeSpan timeout, CancellationToken caller)
    {
        caller.ThrowIfCancellationRequested();
        if (boundary.Console is null && boundary.File is null) { return CaptureFlushResult(boundary); }
        await using var deadline = new FlushDeadline(timeProvider, started, timeout, () => CaptureFlushResult(boundary));
        using var wait = CancellationTokenSource.CreateLinkedTokenSource(caller, deadline.Token);
        try
        {
            await boundary.WaitForWriterFlushAsync(wait.Token).ConfigureAwait(false);
            // A terminal unavailable flush can precede the disposition of an already-started write.
            if (boundary.Console is { } console)
            { await queuedOutputs!.Console!.WaitForCoveredRecordsAsync("console", console, wait.Token).ConfigureAwait(false); }
            if (boundary.File is { } file)
            { await queuedOutputs!.File!.WaitForCoveredRecordsAsync("file", file, wait.Token).ConfigureAwait(false); }
        }
        catch (OperationCanceledException) when (wait.IsCancellationRequested)
        { caller.ThrowIfCancellationRequested(); }
        caller.ThrowIfCancellationRequested();
        return deadline.ObserveCompletion();
    }
    private LogFlushResult CaptureFlushResult(RuntimeFlushBoundary boundary)
    {
        lock (admissionGate)
        {
            var observations = new List<LogOutputDrainResult>(2);
            if (boundary.Console is { } console) { observations.Add(queuedOutputs!.Console!.CaptureDrain("console", console)); }
            if (boundary.File is { } file) { observations.Add(queuedOutputs!.File!.CaptureDrain("file", file)); }
            return new LogFlushResult(boundary.BoundaryCapturedAtUtc, DateTimeOffset.UtcNow,
                boundary.InvalidEventCount, boundary.TruncatedEventCount, observations);
        }
    }
    internal RuntimeFlushBoundary RegisterFlushBoundary()
    {
        if (queuedOutputs is null) { throw new InvalidOperationException("Flush boundaries require actual queued output composition."); }
        lock (admissionGate)
        {
            // Pure system UTC sampling here does not invoke caller factories, payloads or the event clock.
            DateTimeOffset capturedAt = DateTimeOffset.UtcNow;
            OutputFlushBoundary? console = queuedOutputs.Console?.RegisterFlushBoundary();
            OutputFlushBoundary? file = queuedOutputs.File?.RegisterFlushBoundary();
            return new RuntimeFlushBoundary(capturedAt, InvalidEventCount, TruncatedEventCount, console, file);
        }
    }
    internal RuntimeAccounting CaptureAccounting()
    {
        if (queuedOutputs is null) { throw new InvalidOperationException("Accounting requires actual queued output composition."); }
        lock (admissionGate)
        {
            // Output/disposition/truncation/closure changes share this gate. Invalid increments
            // independently, so its atomic read chooses the observation point while the rest is stable.
            return new RuntimeAccounting(admissionClosed, InvalidEventCount, TruncatedEventCount,
                queuedOutputs.Console?.CaptureAccounting(), queuedOutputs.File?.CaptureAccounting());
        }
    }

    /// <summary>Returns an owned immutable live observation without routing a log or performing output I/O.</summary>
    /// <remarks>Available after faults and admission closure. This does not register a fixed flush target.</remarks>
    public CoreLogDiagnostics GetDiagnostics()
    {
        lock (admissionGate)
        {
            RuntimeAccounting counts = CaptureAccounting();
            var values = new List<LogOutputDiagnostics>(2);
            if (counts.Console is OutputAccounting console) { values.Add(new LogOutputDiagnostics("console", console)); }
            if (counts.File is OutputAccounting file) { values.Add(new LogOutputDiagnostics("file", file, recoveredPartialSegment)); }
            CoreLogState state = !counts.AdmissionClosed ? CoreLogState.Running :
                values.All(value => value.State == LogOutputState.Stopped) ? CoreLogState.Stopped : CoreLogState.Stopping;
            return new CoreLogDiagnostics(state, counts.AdmissionClosed, DateTimeOffset.UtcNow, shutdownStartedAtUtc,
                counts.InvalidEventCount, counts.TruncatedEventCount, values, lastStopResult);
        }
    }

    internal OperationScopeStore Scopes { get; } = new();
    internal long InvalidEventCount => Interlocked.Read(ref invalidEventCount);
    internal long TruncatedEventCount => Interlocked.Read(ref truncatedEventCount);

    /// <summary>Creates a logger borrowing this runtime, including after admission closes.</summary>
    /// <exception cref="ArgumentNullException">Component is null.</exception>
    /// <exception cref="ArgumentException">Component is blank, malformed, contains controls or exceeds 255 scalars.</exception>
    public CoreLogger CreateLogger(string component)
    {
        ArgumentNullException.ThrowIfNull(component);
        if (string.IsNullOrWhiteSpace(component) || !UnicodeScalarText.IsValid(component, 255, allowControls: false))

            throw new ArgumentException("Use a component of 1-255 valid Unicode scalars without controls.", nameof(component));

    internal bool IsEnabled(CoreLogLevel level)
    {
        ValidateLevel(level);
        lock (admissionGate)
        {
            return !admissionClosed && outputs.Any(output => Interested(output, level) && !output.Admission.IsFaulted);
        }
    }

    internal LogWriteResult Write(string component, CoreLogLevel level, LogEventData? data, Func<LogEventData>? factory)
    {
        ValidateLevel(level);
        bool renderConsole, renderFile;
        lock (admissionGate)
        {
            if (admissionClosed) { return Stopped(isTruncated: false); }
            if (!outputs.Any(output => Interested(output, level))) { return FinishWithoutCapture(level, LogWriteOutcome.Filtered); }
            if (!outputs.Any(output => Interested(output, level) && !output.Admission.IsFaulted))
            {
                return FinishWithoutCapture(level, LogWriteOutcome.Rejected);
            }
            renderConsole = consoleWriter is not null && outputs.Any(output => output.Name == "console" && Interested(output, level) && !output.Admission.IsFaulted);
            renderFile = fileWriter is not null && outputs.Any(output => output.Name == "file" && Interested(output, level) && !output.Admission.IsFaulted);
        }
        // No gate is held around caller factories, clock, scopes or payload callbacks.
        if (factory is not null) { data = factory(); }
        if (data is null) { return Invalid("InvalidFactoryResult"); }
        string? eventName = data.EventName;
        string? message = data.Message;
        string? template = data.MessageTemplate;
        string? operationId = data.OperationId;
        CapturedEventFacts facts;
        try
        {
            facts = EventProjection.Capture(metadata, Scopes.Capture(), component, level, data, timeProvider)!;
        }
        catch (Exception error) when (error is not (OutOfMemoryException or StackOverflowException))
        {
            return Invalid(CaptureError(eventName, message, template, operationId));
        }
        EncodedEvent captured;
        RenderedEvent? preparedConsole = null, preparedFile = null;
        // Other owned encoding/rendering consistency errors are Core defects, not invalid caller content.
        try
        {
            captured = EventEncoder.Encode(facts, maximumEntryBytes);
            if (renderConsole) { preparedConsole = consoleWriter!.Prepare(captured); }
            if (renderFile) { preparedFile = fileWriter!.Prepare(captured); }
        }
        catch (EventEnvelopeTooLargeException) { return Invalid("EnvelopeTooLarge"); }
        bool isTruncated = captured.IsTruncated || preparedConsole?.IsTruncated == true || preparedFile?.IsTruncated == true;
        lock (admissionGate)
        {
            if (isTruncated) { IncrementLossCounter(ref truncatedEventCount); }
            if (admissionClosed) { return Stopped(isTruncated); }
            return Admit(level, captured, preparedConsole?.Line, preparedFile?.Line, isTruncated);
        }
    }

    // Primitive internal qualification closure: no final flush or bounded shutdown is promised.
    // Supported applications use StopAsync/DisposeAsync; a later flush cannot reopen an exited worker.
    internal void CloseAdmission()
    {
        lock (admissionGate) { shutdownStartedAtUtc ??= DateTimeOffset.UtcNow; admissionClosed = true; queuedOutputs?.Complete(); }
    }

    private LogWriteResult Invalid(string code)
    {
        IncrementLossCounter(ref invalidEventCount);
        return new LogWriteResult(LogWriteOutcome.InvalidEvent, [], code);
    }

    // Loss counters retain a nonnegative lower bound once Int64 can no longer represent the total.
    private static void IncrementLossCounter(ref long counter)
    {
        long current = Interlocked.Read(ref counter);
        while (current != long.MaxValue)
        {
            long observed = Interlocked.CompareExchange(ref counter, current + 1, current);
            if (observed == current) { return; }
            current = observed;
        }
    }

    private LogWriteResult Stopped(bool isTruncated) => new(LogWriteOutcome.Stopped,
        outputs.Select(output => new LogOutputWriteResult(output.Name, LogOutputWriteStatus.Stopped, null)), isTruncated: isTruncated);

    private LogWriteResult FinishWithoutCapture(CoreLogLevel level, LogWriteOutcome outcome)
    {
        var observations = new List<LogOutputWriteResult>(outputs.Length);
        foreach (Output output in outputs)
        {
            LogOutputWriteStatus status;
            if (Interested(output, level))
            {
                output.Admission.RecordUnavailable();
                status = LogOutputWriteStatus.Faulted;
            }
            else
            {
                output.Admission.RecordFiltered();
                status = LogOutputWriteStatus.Filtered;
            }
            observations.Add(new LogOutputWriteResult(output.Name, status, null));
        }
        return new LogWriteResult(outcome, observations);
    }

    private LogWriteResult Admit(CoreLogLevel level, EncodedEvent captured, ReadOnlyMemory<byte>? consoleLine, ReadOnlyMemory<byte>? fileLine, bool isTruncated)
    {
        var observations = new List<LogOutputWriteResult>(outputs.Length);
        int accepted = 0;
        int interested = 0;
        foreach (Output output in outputs)
        {
            long? sequence = null;
            LogOutputWriteStatus status;
            if (!Interested(output, level))
            {
                output.Admission.RecordFiltered();
                status = LogOutputWriteStatus.Filtered;
            }
            else
            {
                interested++;
                if (output.Admission.IsFaulted)
                {
                    output.Admission.RecordUnavailable();
                    status = LogOutputWriteStatus.Faulted;
                }
                else if (output.Name == "console" && consoleLine is { } preparedLine)
                { status = queuedOutputs!.Console!.TryAdmit(captured, preparedLine, out sequence); }
                else if (output.Name == "file" && fileLine is { } preparedFileLine)
                { status = queuedOutputs!.File!.TryAdmit(captured, preparedFileLine, out sequence); }
                else { status = output.Admission.TryAdmit(captured, out sequence); }
                if (status == LogOutputWriteStatus.Accepted) { accepted++; }
            }
            if (Interested(output, level) && status is not (LogOutputWriteStatus.Accepted or LogOutputWriteStatus.QueueFull or LogOutputWriteStatus.Faulted))
            {
                throw new InvalidOperationException("Output admission returned an invalid status.");
            }
            if (status == LogOutputWriteStatus.Accepted ? sequence is null or <= 0 : sequence is not null)
            {
                throw new InvalidOperationException("Output admission returned an invalid ticket.");
            }
            observations.Add(new LogOutputWriteResult(output.Name, status, sequence));
        }
        LogWriteOutcome outcome = LogWriteOutcome.Rejected;
        if (accepted == interested) { outcome = LogWriteOutcome.Accepted; }
        else if (accepted > 0) { outcome = LogWriteOutcome.PartiallyAccepted; }
        return new LogWriteResult(outcome, observations, isTruncated: isTruncated);
    }

    private static string CaptureError(string? name, string? message, string? template, string? operationId)
    {
        foreach (string? text in new[] { name, message, template, operationId })
        {
            if (text is not null && !UnicodeScalarText.IsValid(text, int.MaxValue, allowControls: true)) { return "InvalidUnicode"; }
        }
        if (string.IsNullOrEmpty(name) || !UnicodeScalarText.IsValid(name, 255, allowControls: false)) { return "InvalidEventName"; }
        if (message is null) { return "InvalidMessage"; }
        string? operation = operationId?.Trim();
        if (!string.IsNullOrEmpty(operation) && !UnicodeScalarText.IsValid(operation, 255, allowControls: false)) { return "InvalidOperationId"; }
        // Arbitrary cooperative callback/unsupported payload failures expose only this fixed code.
        return "UnsupportedValue";
    }

    private static bool Interested(Output output, CoreLogLevel level) => level != CoreLogLevel.None && level >= output.Threshold;
    private static void ValidateSettings(CoreLogLevel console, CoreLogLevel file, int maximumBytes)
    {
        ValidateLevel(console); ValidateLevel(file);
        if (maximumBytes is < LoggingOptions.MinimumMaxEntryBytes or > LoggingOptions.MaximumMaxEntryBytes) { throw new ArgumentOutOfRangeException("options"); }
    }
    private static void ValidateLevel(CoreLogLevel level)
    {
        if (!Enum.IsDefined(level)) { throw new ArgumentOutOfRangeException(nameof(level), "Use a declared logging level."); }
    }

        else { using (ExecutionContext.SuppressFlow()) { acquired = Create(); } }

        {
            if (firstAbandonedSequence == 0) { firstAbandonedSequence = abandoned.AcceptedSequence; }
            lastAbandonedSequence = abandoned.AcceptedSequence; abandonedCount++;
        }
        NotifyDispositionChanged();
        foreach (OutputFlushRegistration registration in flushRequests.Values)
        {
            if (!ReferenceEquals(registration, flushing)) { registration.Resolve(false); }
        }
        OutputFlushRegistration? active = flushing;
        flushRequests.Clear();
        // An already-started flush remains unresolved until its actual I/O settles, even if admission faults.
        if (active is not null) { flushRequests.Add(active.AcceptedSequenceBoundary, active); }

            lastFaultCode = code; lastFaultAtUtc = DateTimeOffset.UtcNow;
            state = workerFinished && stopRequested ? LogOutputState.Stopped : stopRequested ? LogOutputState.Stopping : LogOutputState.Faulted;
            foreach (OutputFlushRegistration registration in flushRequests.Values)
            {
                if (!ReferenceEquals(registration, flushing)) { registration.Resolve(false); }
            }
            flushRequests.Clear();
            // Actual flush I/O keeps its own registration until it settles.
            if (flushing is not null) { flushRequests.Add(flushing.AcceptedSequenceBoundary, flushing); }

        }

    private async Task ProcessAsync()
    {
        Exception? primaryFailure = null;
        try
        {
            while (true)
            {
                AdmittedEvent admitted = default;
                OutputFlushRegistration? request = null;
                Task? wait = null;
                lock (admissionGate)
                {
                    if (faulted) { workerExiting = true; return; }
                    if (stopDeadlineExpired)
                    {
                        // Expiry cancels active I/O; it never starts a later flush.
                        foreach (OutputFlushRegistration registration in flushRequests.Values) { registration.Resolve(false); }
                        flushRequests.Clear();
                    }
                    if (flushRequests.Count > 0 && flushRequests.First().Key <= writtenCount)
                    {
                        request = flushRequests.First().Value;
                        flushing = request;
                    }
                    else if (channel.Reader.TryRead(out admitted)) { inFlight = true; inFlightSequence = admitted.AcceptedSequence; }
                    else if (stopRequested) { workerExiting = true; return; }
                    else
                    {
                        // Reset only while no work is visible under the admission gate: no lost wakeup.
                        if (wake.Task.IsCompleted) { wake = NewWake(); }
                        wait = wake.Task;
                    }
                }
                if (wait is not null) { await wait.ConfigureAwait(false); continue; }
                if (request is not null)
                {
                    // Flush at this fixed boundary before starting a later record. All I/O is outside the gate.
                    await flush!(ioCancellation?.Token ?? CancellationToken.None).ConfigureAwait(false);
                    lock (admissionGate)
                    {
                        flushRequests.Remove(request.AcceptedSequenceBoundary);
                        flushing = null;
                        request.Resolve(true);
                    }
                }
                else
                {
                    // Complete record confirmation only; a dequeued/partially emitted write is still pending.
                    await write(admitted, ioCancellation?.Token ?? CancellationToken.None).ConfigureAwait(false);
                    lock (admissionGate) { writtenCount++; inFlight = false; NotifyDispositionChanged(); }
                }
            }
        }
        catch (Exception error)
        {
            primaryFailure = error;
            lock (admissionGate)
            {
                if (inFlight) { failedCount++; failedSequence = inFlightSequence; inFlight = false; }
                bool flushFailed = flushing is not null;
                flushing?.Resolve(false);
                flushing = null;
                Fault(flushFailed ? "FlushFailed" : error is FileOutputWriteException fileError ? fileError.FaultCode : "WriteFailed");
            }
            throw;
        }
        finally
        {
            // No pending writer call remains. Owned cleanup precedes terminal state/Completion,
            // outside the gate, and never receives a borrowed console stream.
            lock (admissionGate) { workerExiting = true; }
            List<Exception>? cleanupFailures = null;
            try { await CloseIoCancellationAsync().ConfigureAwait(false); }
            catch (Exception error) { RecordCleanupFault("CancellationCallbacksFailed"); (cleanupFailures ??= []).Add(error); }
            try { await CancelFileFlushScheduling().ConfigureAwait(false); }
            catch (Exception error) { RecordCleanupFault("FileFlushTimerCleanupFailed"); (cleanupFailures ??= []).Add(error); }
            try { ownedResource?.Dispose(); }
            catch (Exception error) { RecordCleanupFault("OutputCleanupFailed"); (cleanupFailures ??= []).Add(error); }
            lock (admissionGate)
            {
                admissionClosed = true; channel.Writer.TryComplete();
                workerExiting = true; workerFinished = true;
                state = stopRequested || !faulted ? LogOutputState.Stopped : LogOutputState.Faulted;
            }
            if (cleanupFailures is not null)
            {
                // Preserve the exact original I/O failure as well as every failed cleanup step.
                if (primaryFailure is not null) { cleanupFailures.Insert(0, primaryFailure); }
                if (cleanupFailures.Count == 1) { ExceptionDispatchInfo.Capture(cleanupFailures[0]).Throw(); }
                throw new AggregateException("Output cleanup failed.", cleanupFailures);
            }
        }
    }

namespace CIS.Logging.Runtime;

// Creates exactly one independent worker for each enabled destination and no disabled worker.
internal sealed class OutputQueues
{
    private readonly object admissionGate;
    internal OutputQueues(int capacity, bool consoleEnabled, bool fileEnabled,
        Func<AdmittedEvent, ValueTask>? consoleWrite, Func<AdmittedEvent, ValueTask>? fileWrite, object? sharedAdmissionGate = null, Func<Func<Task>, Task>? launchWorker = null, IDisposable? fileResource = null, Func<ValueTask>? consoleFlush = null, Func<ValueTask>? fileFlush = null,
        Func<AdmittedEvent, CancellationToken, ValueTask>? consoleCancellableWrite = null, Func<AdmittedEvent, CancellationToken, ValueTask>? fileCancellableWrite = null,
        Func<CancellationToken, ValueTask>? consoleCancellableFlush = null, Func<CancellationToken, ValueTask>? fileCancellableFlush = null)
    {
        admissionGate = sharedAdmissionGate ?? new object();
        if (capacity is < LoggingOptions.MinimumQueueCapacity or > LoggingOptions.MaximumQueueCapacity) { throw new ArgumentOutOfRangeException(nameof(capacity)); }
        if (consoleEnabled != (consoleWrite is not null)) { throw new ArgumentException("Writer presence must match enabled console output.", nameof(consoleWrite)); }
        if (fileEnabled != (fileWrite is not null)) { throw new ArgumentException("Writer presence must match enabled file output.", nameof(fileWrite)); }
        if (!consoleEnabled && consoleFlush is not null) { throw new ArgumentException("Disabled console output cannot have a flush callback.", nameof(consoleFlush)); }
        if (!fileEnabled && fileFlush is not null) { throw new ArgumentException("Disabled file output cannot have a flush callback.", nameof(fileFlush)); }
        try
        {
            if (consoleWrite is not null) { Console = Create("console", consoleWrite, null, consoleFlush, consoleCancellableWrite, consoleCancellableFlush); }
            if (fileWrite is not null) { File = Create("file", fileWrite, fileResource, fileFlush, fileCancellableWrite, fileCancellableFlush); }
            Completion = Task.WhenAll(Console?.Completion ?? Task.CompletedTask, File?.Completion ?? Task.CompletedTask);
        }
        catch
        {
            // Unpublished queues are empty. Close/join only this attempt's idle workers, outside the gate.
            Complete();
            try { Task.WhenAll(Console?.Completion ?? Task.CompletedTask, File?.Completion ?? Task.CompletedTask).GetAwaiter().GetResult(); } catch { }
            throw;
        }
        OutputEventQueue Create(string name, Func<AdmittedEvent, ValueTask> write, IDisposable? ownedResource, Func<ValueTask>? flush, Func<AdmittedEvent, CancellationToken, ValueTask>? cancellableWrite, Func<CancellationToken, ValueTask>? cancellableFlush)
        {
            try { return new OutputEventQueue(capacity, write, admissionGate, launchWorker, ownedResource, flush, cancellableWrite, cancellableFlush); }
            catch { throw new CoreLogInitializationException(name, "WorkerInitializationFailed"); }
        }
    }
    internal OutputEventQueue? Console { get; }
    internal OutputEventQueue? File { get; }
    internal Task Completion { get; }

    internal void AbandonUnstarted() { lock (admissionGate) { Console?.AbandonUnstarted(); File?.AbandonUnstarted(); } }

}

[TestClass]
public sealed class LoggerFoundationArchitectureTests
{
    [TestMethod]
    public void ImplementedLoggerMethodsHaveRealBodiesAndNamedParameters()
    {
        Assert.IsTrue(typeof(CoreLogger).IsSealed);
        Assert.IsTrue(typeof(CoreLogRuntime).IsSealed);
        Assert.AreEqual(0, typeof(CoreLogger).GetConstructors().Length);
        Assert.AreEqual(0, typeof(CoreLogRuntime).GetConstructors().Length);
        var methods = typeof(CoreLogger).GetMethods(BindingFlags.Public | BindingFlags.Instance | BindingFlags.DeclaredOnly);
        CollectionAssert.AreEquivalent(new[] { "IsEnabled", "Write", "WriteLazy" }, methods.Select(method => method.Name).ToArray());
        var runtimeMethods = typeof(CoreLogRuntime).GetMethods(BindingFlags.Public | BindingFlags.Instance | BindingFlags.Static | BindingFlags.DeclaredOnly);
        foreach (MethodInfo method in methods.Concat(runtimeMethods))
        {
            Assert.IsNotNull(method.GetMethodBody());
            Assert.IsFalse(method.MethodImplementationFlags.HasFlag(MethodImplAttributes.InternalCall));
        }
        CollectionAssert.AreEqual(new[] { "level", "data" }, typeof(CoreLogger).GetMethod("Write")!.GetParameters().Select(item => item.Name).ToArray());
        CollectionAssert.AreEqual(new[] { "level", "factory" }, typeof(CoreLogger).GetMethod("WriteLazy")!.GetParameters().Select(item => item.Name).ToArray());
        CollectionAssert.AreEquivalent(new[] { "Create", "CreateLogger", "GetDiagnostics", "FlushAsync", "StopAsync", "DisposeAsync" }, runtimeMethods.Select(method => method.Name).ToArray());
        CollectionAssert.AreEqual(new[] { "cancellationToken" }, typeof(CoreLogRuntime).GetMethod("StopAsync")!.GetParameters().Select(item => item.Name).ToArray());
        Assert.AreEqual(typeof(CancellationToken), typeof(CoreLogRuntime).GetMethod("StopAsync")!.GetParameters()[0].ParameterType);
        Assert.AreEqual(typeof(Task<LogStopResult>), typeof(CoreLogRuntime).GetMethod("StopAsync")!.ReturnType);
        Assert.IsTrue(typeof(IAsyncDisposable).IsAssignableFrom(typeof(CoreLogRuntime)));
        Assert.AreEqual(typeof(ValueTask), typeof(CoreLogRuntime).GetMethod("DisposeAsync")!.ReturnType);
        Assert.AreEqual(0, typeof(CoreLogRuntime).GetMethod("DisposeAsync")!.GetParameters().Length);
        Assert.IsNull(typeof(LogEventData).GetProperty("Level"));
        Assert.IsNull(typeof(LogEventData).GetProperty("Severity"));
    }

    [TestMethod]
    public void ResultMembersAreReadonlyAndEnumNumbersMatchThePreservedContract()
    {
        foreach (Type type in new[] { typeof(LogWriteResult), typeof(LogOutputWriteResult), typeof(LogStopResult) })
        {
            Assert.IsTrue(type.IsSealed);
            Assert.AreEqual(0, type.GetConstructors().Length);
            foreach (PropertyInfo property in type.GetProperties()) { Assert.IsNull(property.GetSetMethod(nonPublic: true)); }
        }
        CollectionAssert.AreEquivalent(new[] { "Completion", "Succeeded", "LossObserved", "BoundaryCapturedAtUtc", "ObservedAtUtc", "InvalidEventCount", "TruncatedEventCount", "Outputs" },
            typeof(LogStopResult).GetProperties().Select(property => property.Name).ToArray());
        string root = EventContractTestsRoot();
        using JsonDocument contract = JsonDocument.Parse(File.ReadAllText(Path.Combine(root, "contracts/logging/v1/runtime-contract.json")));
        foreach ((Type type, string section) in new[] { (typeof(LogWriteOutcome), "outcomes"), (typeof(LogOutputWriteStatus), "outputStatuses") })
        {
            foreach (JsonProperty property in contract.RootElement.GetProperty(section).EnumerateObject()) { Assert.AreEqual(property.Value.GetInt32(), Convert.ToInt32(Enum.Parse(type, property.Name))); }
        }
    }

    [TestMethod]
    public void OutputAdmissionInterfaceIsInternalAndReceivesOnlyOwnedEncodedEvents()
    {
        Type seam = typeof(CoreLogger).Assembly.GetType("CIS.Logging.Runtime.ILogOutputAdmission")!;
        Assert.IsTrue(seam.IsInterface);
        Assert.IsFalse(seam.IsPublic);
        Type payload = seam.GetMethod("TryAdmit")!.GetParameters()[0].ParameterType;
        Assert.AreEqual("CIS.Logging.Events.EncodedEvent", payload.FullName);
        Assert.IsFalse(payload.IsPublic);
    }

        StopCleanupExpectations.Validate(mode, report);

    [TestMethod]
    [DataRow(false)]
    [DataRow(true)]
    public void StartupKeepsItsSafeFailureWhenWorkerCleanupThrows(bool timerFailure)
    {
        using var root = new TemporaryDirectory(); using var console = new MemoryStream();
        var options = new LoggingOptions { ConsoleLevel = CoreLogLevel.Information, FileLevel = CoreLogLevel.Information,
            FilePath = Path.Combine(root.Path, "active.jsonl"), MaxEntryBytes = 16384 };
        int launched = 0, settled = 0;
        Task Launch(Func<Task> worker)
        {
            if (Interlocked.Increment(ref launched) == 2 && !timerFailure) { throw new IOException("launch-private"); }
            return Task.Run(async () => { await worker(); Interlocked.Increment(ref settled); throw new IOException("cleanup-private"); });
        }
        CoreLogInitializationException error = Assert.ThrowsExactly<CoreLogInitializationException>(() =>
        {
            if (timerFailure) { _ = CoreLogRuntime.CreateConfiguredQueued(ConsoleCompositionFixture.Metadata(), options, new FailingTimerClock(), _ => console, Launch); }
            else { _ = CoreLogRuntime.CreateConsoleQueued(ConsoleCompositionFixture.Metadata(), options, _ => ValueTask.CompletedTask, acquireConsole: _ => console, launchWorker: Launch); }
        });
        Assert.AreEqual("file", error.OutputName);
        Assert.AreEqual(timerFailure ? "FileFlushTimerInitializationFailed" : "WorkerInitializationFailed", error.ErrorCode);
        Assert.AreEqual(timerFailure ? 2 : 1, settled); Assert.IsTrue(console.CanWrite);
        if (timerFailure)
        {
            using var lease = new FileStream(options.FilePath + ".cis.lock", FileMode.Open, FileAccess.ReadWrite, FileShare.None);
            Assert.AreEqual(0L, new FileInfo(options.FilePath).Length);
        }
    }
    private sealed class FailingTimerClock : TimeProvider
    {
        public override ITimer CreateTimer(TimerCallback callback, object? state, TimeSpan dueTime, TimeSpan period) => throw new IOException("timer-private");
    }
}

    {

        var environment = System.Environment.GetEnvironmentVariables().Keys.Cast<string>()
            .Where(key => key.StartsWith("CORE_", StringComparison.Ordinal)).ToDictionary(key => key, _ => (string?)null, StringComparer.Ordinal);
        string host = Path.Combine(AppContext.BaseDirectory, "discovery-fixture", "CIS.Core.DiscoveryFixture.dll");

        JsonObject report = JsonNode.Parse(child.StandardOutput)!.AsObject();

        Assert.AreEqual(Hash(Path.Combine(Path.GetDirectoryName(host)!, "CIS.Core.dll")), report["coreSha256"]!.GetValue<string>());

# Runtime creation and eventual cleanup

CoreLogRuntime.Create() reads the existing static EnvironmentOptions facts and module settings, prepares the selected logging path against the current directory and normalizes EnvironmentOptions.Logging.FilePath in place, and creates the real configured queues and destinations. Configure EnvironmentOptions.Logging before creation; concurrent mutation remains unsupported. Creation validates options and metadata before output acquisition. The application owns the runtime. Public creation always supplies actual queues, including the disabled-output case; admission-only qualification seams remain internal.

StopAsync and DisposeAsync keep the original shared operation and deadline. An unresolved write or flush retains its owned resource until that I/O settles. Output-owned cancellation callbacks join before owned resources close. Closure or fault starts periodic-timer disposal once. The worker joins cancellation callbacks and that shared timer-disposal task, then attempts owned-resource cleanup once, outside admission synchronization. A throwing inactive Dispose is not retried. Borrowed console streams remain open. Successful physical file disposal closes the active handle and group lease; pre-existing records are preserved.

If I/O and cleanup both fail, the internal worker completion retains the original I/O exception first and all cleanup failures. A lone cleanup-step exception keeps its original type; a throwing cancellation callback retains the AggregateException supplied by CancellationTokenSource.CancelAsync. Public stop converts terminal output failures into its existing safe delivery result. LastFaultCode and LastFaultAtUtc report the latest fixed safe category, including CancellationCallbacksFailed, FileFlushTimerCleanupFailed, OutputCleanupFailed and ShutdownTimerCleanupFailed, without exception payloads. Cleanup can supersede an earlier delivery category; a shutdown-timer category applies to every enabled output. Original stop results and delivery counts remain the delivery record. A cleanup fault does not reclassify a delivered record. Stop Succeeded concerns records and writer flushing; resource cleanup is a separate observation. A late deadline-timer disposal fault updates live diagnostics after publication and never changes the cached LastStopResult. No new public diagnostic fields or cleanup framework are added.

FlushAsync does not reopen a closed writer. After successful stop, a new flush may report unavailable/Failed and LossObserved for its own failed confirmation; that does not rewrite the earlier successful stop acknowledgement. The internal CloseAdmission helper is primitive qualification closure without a final flush/deadline. Applications use StopAsync or DisposeAsync. A stop setup failure settles its already-published task; the catastrophic allocation/scheduling trigger is source-inspected, rather than falsely claimed as a controlled native execution.

Thirteen controlled kernel cases cover owned disposal after held I/O, primary write/flush plus cleanup errors, timer/resource-only errors, throwing/held cancellation callbacks, late timer cleanup before or after held I/O settles, repeated disposal, closed flushing and primitive closure. Two distinguishing unit cases also retain safe startup categories through terminal worker-cleanup errors and verify physical ownership release. Five fresh-process public creation cases use actual file ownership, disabled outputs, invalid settings/metadata and contention. These tests do not substitute for T05's complete physical/process races and abrupt termination, or full release qualification. Exact executed evidence and review status are recorded in the task receipt. FlushAsync keeps its existing exception behavior for a throwing custom deadline-timer disposer; token-source cleanup runs in finally. The public factory uses the system clock.

using System.Runtime.ExceptionServices;

namespace CIS.Logging.Runtime;

// One timer per active caller, with bounded slices so every positive TimeSpan is supported.
// Lock order is deadline gate -> admission gate through observe; no inverse acquisition exists.
internal sealed class FlushDeadline : IAsyncDisposable
{
    private readonly object gate = new();
    private readonly TimeProvider clock;
    private readonly Func<LogFlushResult> observe;
    private readonly Func<LogFlushResult>? observeDeadline;
    private readonly CancellationTokenSource lifetime = new();
    private readonly ITimer timer;
    private long remainingTicks;
    private long checkpoint;
    private bool disposed;
    private LogFlushResult? observation;
    private ExceptionDispatchInfo? error;

    internal FlushDeadline(TimeProvider clock, long started, TimeSpan timeout, Func<LogFlushResult> observe, Func<LogFlushResult>? observeDeadline = null)
    {
        this.clock = clock; this.observe = observe; this.observeDeadline = observeDeadline; remainingTicks = timeout.Ticks; checkpoint = started;
        timer = clock.CreateTimer(static state => ((FlushDeadline)state!).Tick(), this, Timeout.InfiniteTimeSpan, Timeout.InfiniteTimeSpan);
        Tick();
    }
    internal CancellationToken Token => lifetime.Token;

    private void UpdateRemaining()
    {
        long now = clock.GetTimestamp();
        long elapsed = clock.GetElapsedTime(checkpoint, now).Ticks;
        if (elapsed > 0) { remainingTicks -= Math.Min(remainingTicks, elapsed); }
        checkpoint = now;
    }
    private void Tick()
    {
        bool expire = false;
        lock (gate)
        {
            if (disposed || observation is not null || error is not null) { return; }
            try
            {
                UpdateRemaining();
                if (remainingTicks == 0) { observation = (observeDeadline ?? observe)(); expire = true; }
                else
                {
                    // Timer resolution may overshoot sub-millisecond budgets; never expire early.
                    long delay = Math.Clamp(remainingTicks, TimeSpan.TicksPerMillisecond, TimeSpan.TicksPerDay);
                    timer.Change(TimeSpan.FromTicks(delay), Timeout.InfiniteTimeSpan);
                }
            }
            catch (Exception failure) when (failure is not (OutOfMemoryException or StackOverflowException))
            { error = ExceptionDispatchInfo.Capture(failure); expire = true; }
        }
        // Lifetime cancellation runs outside both gates. A stop observer may already have
        // requested asynchronous output cancellation under deadline -> admission lock order.
        if (expire) { lifetime.Cancel(); }
    }
    internal LogFlushResult ObserveCompletion()
    {
        lock (gate)
        {
            error?.Throw();
            // A deadline observation wins over later I/O; do not replace its pending facts.
            return observation ??= observe();
        }
    }
    public async ValueTask DisposeAsync()
    {
        lock (gate) { disposed = true; }
        // Join an active callback before disposing the token source it may still cancel.
        try { await timer.DisposeAsync().ConfigureAwait(false); }
        finally { lifetime.Dispose(); }
    }
}

using CIS.Logging.Events;
using CIS.Logging.Rendering;

namespace CIS.Logging.Runtime;

// Only the sequential file worker writes. Its terminal path releases this owned resource.
internal sealed class FileOutputWriter(string absolutePath, LogOutputFormat format, HumanOutputDetail detail, int maximumEntryBytes, long maximumFileBytes, int maximumRetainedFiles) : IDisposable
{
    private FileOutputOwner? owner;
    private long activeBytes;
    internal bool RecoveredPartialSegment { get; private set; }
    internal void Initialize()
    {
        if (maximumRetainedFiles is < LoggingOptions.MinimumFileRetainedFiles or > LoggingOptions.MaximumFileRetainedFiles)
        { throw new ArgumentOutOfRangeException(nameof(maximumRetainedFiles)); }
        FileOutputOwner acquired = FileOutputOwner.Initialize(absolutePath, format, maximumRetainedFiles, maximumFileBytes);
        try
        {
            RecoveredPartialSegment = acquired.RecoveredPartialSegment;
            long length = acquired.ActiveFile.Length;
            if (length > maximumFileBytes) { throw new CoreLogInitializationException("file", "FileSizeLimitExceeded"); }
            bool partial = false;
            if (format == LogOutputFormat.JsonLines)
            {
                JsonlRecordBoundary boundary = JsonlRecordBoundary.Inspect(acquired);
                if (boundary.Status == JsonlBoundaryStatus.Invalid)
                { throw new CoreLogInitializationException("file", "FileRecordBoundaryInvalid"); }
                partial = boundary.Status == JsonlBoundaryStatus.Partial;
            }
            else if (length != 0)
            {
                acquired.ActiveFile.Position = length - 1;
                if (acquired.ActiveFile.ReadByte() != (byte)'\n') { throw new CoreLogInitializationException("file", "FileRecordBoundaryInvalid"); }
            }
            // Failed startup preserves every pre-existing retained segment. A lower
            // selected bound needs explicit archival or a new group before retry.
            if (acquired.Inventory!.Segments.Count - 1 > maximumRetainedFiles)
            { throw new CoreLogInitializationException("file", "FileRetentionLimitExceeded"); }
            if (partial)
            {
                acquired.RecoverPartialSegment(maximumRetainedFiles);
                length = 0;
                RecoveredPartialSegment = true;
            }
            acquired.ActiveFile.Position = length;
            activeBytes = length;
            owner = acquired;
        }
        catch (CoreLogInitializationException) { acquired.Dispose(); throw; }
        catch { acquired.Dispose(); throw new CoreLogInitializationException("file", "FileInitializationFailed"); }
    }
    internal RenderedEvent Prepare(EncodedEvent captured) => EventRenderer.Render(captured, format, detail, maximumEntryBytes);
    internal ValueTask FlushAsync() => FlushAsync(CancellationToken.None);
    internal ValueTask FlushAsync(CancellationToken cancellationToken) => new((owner ?? throw new InvalidOperationException("File output is not initialized.")).ActiveFile.FlushAsync(cancellationToken));
    internal ValueTask WriteAsync(AdmittedEvent admitted) => WriteAsync(admitted, CancellationToken.None);
    internal async ValueTask WriteAsync(AdmittedEvent admitted, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        FileOutputOwner active = owner ?? throw new InvalidOperationException("File output is not initialized.");
        ReadOnlyMemory<byte> line = admitted.PreparedLine ?? throw new InvalidOperationException("File records require prepared output bytes.");
        // The prepared record includes every encoded byte and its LF. Never split it.
        if (line.Length > maximumFileBytes)
        {
            if (format == LogOutputFormat.Human) { throw FileOutputWriteException.RenderTooLarge(); }
            throw new IOException("File record exceeds the segment byte limit.");
        }
        if (line.Length > maximumFileBytes - activeBytes)
        {
            active.Rotate(maximumRetainedFiles);
            activeBytes = 0;
        }
        await active.ActiveFile.WriteAsync(line, cancellationToken).ConfigureAwait(false);
        activeBytes += line.Length;
    }
    public void Dispose() => Interlocked.Exchange(ref owner, null)?.Dispose();
}

namespace CIS.Logging.Runtime;

// Acquisition owns physical handles only. Inventory validation must precede any append.
internal sealed class FileOutputOwner : IDisposable
{
    private readonly FileStream lease;
    internal string AbsolutePath { get; }
    internal FileStream ActiveFile { get; private set; }
    internal FileGroupInventory? Inventory { get; private set; }
    internal bool ActiveWasCreated { get; }
    internal bool RecoveredPartialSegment { get; private set; }

    private FileOutputOwner(string absolutePath, FileStream lease, FileStream activeFile, bool activeWasCreated)
    {
        AbsolutePath = absolutePath;
        this.lease = lease;
        ActiveFile = activeFile;
        ActiveWasCreated = activeWasCreated;
    }

    internal static FileOutputOwner Initialize(string capturedAbsolutePath) => InitializeCore(capturedAbsolutePath, rejectLostActive: false);

    private static FileOutputOwner InitializeCore(string capturedAbsolutePath, bool rejectLostActive, LogOutputFormat? format = null, int maximumRetainedFiles = LoggingOptions.MaximumFileRetainedFiles, long maximumFileBytes = long.MaxValue)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(capturedAbsolutePath);
        if (!Path.IsPathFullyQualified(capturedAbsolutePath))
        {
            throw new ArgumentException("File output requires a captured absolute path.", nameof(capturedAbsolutePath));
        }

        FileStream? lease = null, active = null;
        try
        {
            Directory.CreateDirectory(Path.GetDirectoryName(capturedAbsolutePath)!);
            string leasePath = capturedAbsolutePath + ".cis.lock";
            try
            {
                lease = new FileStream(leasePath, FileMode.OpenOrCreate, FileAccess.ReadWrite, FileShare.None);
            }
            catch (IOException)
            {
                throw new CoreLogInitializationException("file", "FileOwnershipUnavailable");
            }

            // A reserved lease is empty and persistent. Never truncate an existing collision.
            if (lease.Length != 0 || !ExcludesSecondHandle(leasePath))
            {
                throw new CoreLogInitializationException("file", "FileOwnershipUnavailable");
            }
            bool recoveredPartial = format is { } selected && FileGroupTransition.Reconcile(capturedAbsolutePath, selected, maximumRetainedFiles, maximumFileBytes);
            bool activeWasCreated = false;
            try { active = OpenActive(capturedAbsolutePath, FileMode.Open); }
            catch (FileNotFoundException)
            {
                if (rejectLostActive && HasExistingInventory(capturedAbsolutePath)) { throw new CoreLogInitializationException("file", "FileInventoryIncompatible"); }
                active = OpenActive(capturedAbsolutePath, FileMode.CreateNew); activeWasCreated = true;
            }
            var owner = new FileOutputOwner(capturedAbsolutePath, lease, active, activeWasCreated) { RecoveredPartialSegment = recoveredPartial };
            lease = null;
            active = null;
            return owner;
        }
        catch (Exception error) when (error is IOException or UnauthorizedAccessException or System.Security.SecurityException or ArgumentException or NotSupportedException)
        {
            throw new CoreLogInitializationException("file", "FileInitializationFailed");
        }
        finally
        {
            try { active?.Dispose(); }
            finally { lease?.Dispose(); }
        }
    }

    private static bool HasExistingInventory(string path)
    {
        try
        {
            using var existing = new FileStream(path + ".cis.inventory.json", FileMode.Open, FileAccess.Read, FileShare.Read);
            return true;
        }
        catch (FileNotFoundException) { return false; }
        catch (Exception error) when (error is IOException or UnauthorizedAccessException or System.Security.SecurityException)
        {
            throw new CoreLogInitializationException("file", "FileInventoryInitializationFailed");
        }
    }

    private static FileStream OpenActive(string path, FileMode mode)
        => new(path, mode, FileAccess.ReadWrite, FileShare.Read, bufferSize: 1, FileOptions.Asynchronous | FileOptions.SequentialScan);

    internal static FileOutputOwner Initialize(string capturedAbsolutePath, LogOutputFormat format, int maximumRetainedFiles = LoggingOptions.MaximumFileRetainedFiles, long maximumFileBytes = long.MaxValue)
    {
        if (!Enum.IsDefined(format)) { throw new ArgumentOutOfRangeException(nameof(format)); }
        FileOutputOwner owner = InitializeCore(capturedAbsolutePath, rejectLostActive: true, format, maximumRetainedFiles, maximumFileBytes);
        try
        {
            owner.Inventory = FileGroupInventory.OpenOrCreate(owner, format);
            return owner;
        }
        catch { owner.Dispose(); throw; }
    }

    internal void Rotate(int maximumRetainedFiles)
    {
        RequireOwnership();
        FileGroupInventory current = Inventory ?? throw new InvalidOperationException("Rotation requires a managed inventory.");
        FileGroupInventory next = current.AfterRotation(AbsolutePath, maximumRetainedFiles);
        string directory = Path.GetDirectoryName(AbsolutePath)!;
        string closedPath = Path.Combine(directory, next.Segments[^2].FileName);
        // Reject a known reserved-name collision before retiring any listed evidence.
        // The non-overwriting move below still protects against later external races.
        if (File.Exists(closedPath) || Directory.Exists(closedPath)) { throw new IOException("Reserved closed segment already exists."); }
        // Remove only exact oldest inventory members, while holding the group lease,
        // before the old active file becomes an additional retained segment.
        int removed = current.Segments.Count + 1 - next.Segments.Count;
        try
        {
            for (int index = 0; index < removed; index++) { File.Delete(Path.Combine(directory, current.Segments[index].FileName)); }
        }
        catch (Exception error) when (error is IOException or UnauthorizedAccessException or System.Security.SecurityException)
        {
            // No new retained segment has been created. The sequential worker faults,
            // preserves other outputs and releases this owner's handles without retry.
            throw FileOutputWriteException.RetentionDeleteFailed();
        }
        FileGroupTransition.Begin(this, next, maximumRetainedFiles, recovery: false);
        // The lease remains held throughout. No incoming record is written until the new
        // identity is published. Failure after rename leaves no active path to append to;
        // the pending transition permits a later owner to reconcile verified stages.
        // Any rotation exception makes this writer unusable; its worker faults terminally.
        ActiveFile.Dispose();
        File.Move(AbsolutePath, closedPath, overwrite: false);
        next.PublishReplacement(AbsolutePath);
        Inventory = next;
        ActiveFile = OpenActive(AbsolutePath, FileMode.CreateNew);
        FileGroupTransition.Complete(AbsolutePath);
    }

    internal void RecoverPartialSegment(int maximumRetainedFiles)
    {
        RequireOwnership();
        FileGroupInventory current = Inventory ?? throw new InvalidOperationException("Recovery requires a managed inventory.");
        if (JsonlRecordBoundary.Inspect(this).Status != JsonlBoundaryStatus.Partial)
        { throw new InvalidOperationException("Recovery requires a partial final JSONL record."); }
        FileGroupInventory next = current.AfterPartialRecovery(AbsolutePath, maximumRetainedFiles);
        string damagedPath = Path.Combine(Path.GetDirectoryName(AbsolutePath)!, next.Segments[^2].FileName);
        if (File.Exists(damagedPath) || Directory.Exists(damagedPath))
        { throw new CoreLogInitializationException("file", "FileRecoveryCollision"); }
        FileGroupTransition.Begin(this, next, maximumRetainedFiles, recovery: true);
        // Preserve the entire segment, including any intact earlier records, under
        // the same lease. A failed transition never makes old bytes appendable.
        ActiveFile.Dispose();
        File.Move(AbsolutePath, damagedPath, overwrite: false);
        next.PublishReplacement(AbsolutePath);
        Inventory = next;
        ActiveFile = OpenActive(AbsolutePath, FileMode.CreateNew);
        FileGroupTransition.Complete(AbsolutePath);
        RecoveredPartialSegment = true;
    }

    internal void RequireOwnership()
    {
        if (!lease.CanWrite || !ActiveFile.CanWrite) { throw new InvalidOperationException("File ownership is not active."); }
    }

    private static bool ExcludesSecondHandle(string leasePath)
    {
        try
        {
            using var second = new FileStream(leasePath, FileMode.Open, FileAccess.ReadWrite, FileShare.None);
            return false;
        }
        catch (IOException) { return true; }
    }

    public void Dispose()
    {
        try { ActiveFile.Dispose(); }
        finally { lease.Dispose(); }
        // Do not unlink the lease: a new inode could bypass an existing Unix owner's lock.
    }
}

using CIS.Configuration;
using CIS.Text;

namespace CIS.Logging;

/// <summary>Immutable allowlisted application/process facts, excluding settings, paths and operation context.</summary>
public sealed class EventMetadata
{
    /// <summary>Logging captures the automatically populated static facts without retaining module settings.</summary>
    public EventMetadata()
        : this(EnvironmentOptions.ApplicationName, EnvironmentOptions.BuildVersion, EnvironmentOptions.ServiceVersion,
            EnvironmentOptions.EnvironmentName, EnvironmentOptions.EnvironmentKind, EnvironmentOptions.InstanceId,
            EnvironmentOptions.ProcessStartedAtUtc, EnvironmentOptions.InitializedAtUtc, EnvironmentOptions.OperatingSystem,
            EnvironmentOptions.ProcessId, EnvironmentOptions.CommitId, EnvironmentOptions.BuildId, EnvironmentOptions.HostName,
            EnvironmentOptions.OperatingSystemDescription, EnvironmentOptions.RuntimeDescription,
            EnvironmentOptions.OSArchitecture, EnvironmentOptions.ProcessArchitecture)
    {
    }

    // Direct fact capture also supports internal projection qualification without writable runtime facts.
    internal EventMetadata(string applicationName,
        string buildVersion,
        string serviceVersion,
        string environmentName,
        EnvironmentKind environmentKind,
        string instanceId,
        DateTimeOffset? processStartedAtUtc,
        DateTimeOffset initializedAtUtc,
        string operatingSystem,
        int? processId,
        string? commitId,
        string? buildId,
        string? hostName,
        string? operatingSystemDescription,
        string? runtimeDescription,
        string? osArchitecture,
        string? processArchitecture)
    {
        ApplicationName = applicationName;
        BuildVersion = buildVersion;
        ServiceVersion = serviceVersion;
        EnvironmentName = environmentName;
        EnvironmentKind = environmentKind;
        InstanceId = instanceId;
        ProcessStartedAtUtc = processStartedAtUtc?.ToUniversalTime();
        InitializedAtUtc = initializedAtUtc.ToUniversalTime();
        OperatingSystem = operatingSystem;
        ProcessId = processId;
        CommitId = commitId;
        BuildId = buildId;
        var issues = new List<ConfigurationIssue>();
        HostName = OptionalText(hostName, nameof(HostName), issues);
        OperatingSystemDescription = OptionalText(operatingSystemDescription, nameof(OperatingSystemDescription), issues);
        RuntimeDescription = OptionalText(runtimeDescription, nameof(RuntimeDescription), issues);
        OSArchitecture = ArchitectureText(osArchitecture, nameof(OSArchitecture), issues);
        ProcessArchitecture = ArchitectureText(processArchitecture, nameof(ProcessArchitecture), issues);
        OperatingSystem = OperatingSystem?.Trim().ToLowerInvariant() ?? string.Empty;
        if (OperatingSystem is not ("windows" or "linux" or "macos" or "unknown"))
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidIdentity, "EnvironmentOptions.OperatingSystem", "Use windows, linux, macos or unknown."));
        }
        if (ProcessId is <= 0)
        {
            issues.Add(new(ConfigurationIssueCodes.OutOfRange, "EnvironmentOptions.ProcessId", "Supply a positive process identifier or null."));
        }
        if (InitializedAtUtc == default)
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidIdentity, "EnvironmentOptions.InitializedAtUtc", "Supply the application startup timestamp before using logging metadata."));
        }
        CheckIdentity(ApplicationName, "ApplicationName", 255, issues);
        CheckIdentity(EnvironmentName, "EnvironmentName", 255, issues);
        CheckIdentity(ServiceVersion, "ServiceVersion", 512, issues);
        CheckIdentity(BuildVersion, "BuildVersion", 512, issues);
        if (CommitId is not null)
        {
            CheckIdentity(CommitId, "CommitId", 512, issues);
        }
        if (BuildId is not null)
        {
            CheckIdentity(BuildId, "BuildId", 512, issues);
        }
        if (!Enum.IsDefined(EnvironmentKind))
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidEnvironmentKind, "EnvironmentOptions.EnvironmentKind", "Use a declared environment kind."));
        }
        if (!Guid.TryParseExact(InstanceId, "D", out Guid instance) || instance == Guid.Empty || InstanceId.Length != 36)
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidIdentity, "EnvironmentOptions.InstanceId", "Supply a nonempty UUID in D format."));
        }
        else
        {
            InstanceId = instance.ToString("D");
        }
        if (issues.Count != 0)
        {
            throw new CoreConfigurationException(issues);
        }
    }

    /// <summary>The application identity, independent of collector destinations.</summary>
    public string ApplicationName { get; }

    /// <summary>The actual application build identity, including prerelease and build text.</summary>
    public string BuildVersion { get; }

    /// <summary>The deployment or telemetry release label, independent of BuildVersion.</summary>

    /// <summary>The application environment label; it does not modify the host environment.</summary>
    public string EnvironmentName { get; }

    /// <summary>The explicitly selected application execution kind.</summary>
    public EnvironmentKind EnvironmentKind { get; }

    /// <summary>The application instance UUID generated by the container.</summary>

    /// <summary>The actual process start instant in UTC, or null when unavailable.</summary>
    public DateTimeOffset? ProcessStartedAtUtc { get; }

    /// <summary>The container construction instant in UTC.</summary>
    public DateTimeOffset InitializedAtUtc { get; }

    /// <summary>The host name observed by this application, or null when unavailable.</summary>
    public string? HostName { get; }

    /// <summary>The lowercase operating-system family.</summary>
    public string OperatingSystem { get; }

    /// <summary>The observed operating-system description, or null when unavailable.</summary>
    public string? OperatingSystemDescription { get; }

    /// <summary>The lowercase operating-system architecture, or null when unavailable.</summary>
    public string? OSArchitecture { get; }

    /// <summary>The lowercase process architecture, or null when unavailable.</summary>
    public string? ProcessArchitecture { get; }

    /// <summary>The observed runtime description, or null when unavailable.</summary>
    public string? RuntimeDescription { get; }

    /// <summary>The observed positive process identifier, or null when unavailable.</summary>
    public int? ProcessId { get; }

    /// <summary>An optional explicit build commit identifier.</summary>
    public string? CommitId { get; }

    /// <summary>An optional explicit build identifier.</summary>
    public string? BuildId { get; }

    private static string? OptionalText(string? value, string name, List<ConfigurationIssue> issues)
    {
        if (string.IsNullOrWhiteSpace(value))
        {
            return null;
        }
        value = value.Trim();
        if (!UnicodeScalarText.IsValid(value, int.MaxValue, allowControls: true))
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidIdentity, "EnvironmentOptions." + name, "Supply valid Unicode text or omit the optional fact."));
            return null;
        }
        return value;
    }

    private static string? ArchitectureText(string? value, string name, List<ConfigurationIssue> issues)
    {
        value = OptionalText(value, name, issues)?.ToLowerInvariant();
        if (value is not null && value.Any(character => !(character is >= 'a' and <= 'z' or >= '0' and <= '9' or '_')))
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidIdentity, "EnvironmentOptions." + name, "Use lowercase letters, digits and underscores for architecture text."));
        }
        return value;
    }

    private static void CheckIdentity(string? value, string name, int maximum, List<ConfigurationIssue> issues)
    {
        if (string.IsNullOrWhiteSpace(value) || !UnicodeScalarText.IsValid(value, maximum, allowControls: false))
        {
            string code = name == "ApplicationName" && string.IsNullOrWhiteSpace(value)
                ? ConfigurationIssueCodes.MissingApplicationName : ConfigurationIssueCodes.InvalidIdentity;
            issues.Add(new(code, "EnvironmentOptions." + name, "Supply a nonblank bounded identity without controls or malformed Unicode."));
        }
    }
}

using CIS.Configuration;
using CIS.Logging.Configuration;

namespace CIS.Logging;

/// <summary>Caller-owned logging values and defaults. Setters do not validate or start outputs.</summary>
public sealed class LoggingOptions
{
    internal const int MinimumQueueCapacity = 1;
    internal const int MaximumQueueCapacity = 65536;
    internal const int MinimumFlushIntervalMilliseconds = 10;
    internal const int MaximumFlushIntervalMilliseconds = 60000;
    internal const int MinimumShutdownTimeoutMilliseconds = 1;
    internal const int MaximumShutdownTimeoutMilliseconds = 300000;
    internal const int MinimumFileMaxBytes = 65536;
    internal const int MaximumFileMaxBytes = 1073741824;
    internal const int MinimumFileRetainedFiles = 1;
    internal const int MaximumFileRetainedFiles = 1000;
    internal const int MinimumMaxEntryBytes = 16384;
    internal const int MaximumMaxEntryBytes = 16777216;

    /// <summary>The Level setting.</summary>
    public CoreLogLevel Level { get; set; } = CoreLogLevel.Information;

    /// <summary>The ConsoleFormat setting.</summary>
    public LogOutputFormat ConsoleFormat { get; set; } = LogOutputFormat.Human;

    /// <summary>The ConsoleLevel setting (null inherits Level).</summary>
    public CoreLogLevel? ConsoleLevel { get; set; }

    /// <summary>The ConsoleStream setting.</summary>
    public ConsoleOutputStream ConsoleStream { get; set; } = ConsoleOutputStream.StandardError;

    /// <summary>The HumanDetail setting.</summary>
    public HumanOutputDetail HumanDetail { get; set; } = HumanOutputDetail.Normal;

    /// <summary>The FilePath setting.</summary>
    public string? FilePath { get; set; }

    /// <summary>The FileFormat setting.</summary>
    public LogOutputFormat FileFormat { get; set; } = LogOutputFormat.JsonLines;

    /// <summary>The FileLevel setting (null inherits Level).</summary>
    public CoreLogLevel? FileLevel { get; set; }

    /// <summary>The QueueCapacity setting.</summary>
    public int QueueCapacity { get; set; } = 1024;

    /// <summary>Target interval in milliseconds for scheduling file-buffer flushes, including idle periods.</summary>
    /// <remarks>
    /// Defaults to 1,000 ms; Prepare requires 10–60,000 ms.
    /// CORE_LOG_FLUSH_INTERVAL_MS populates this value in EnvironmentOptions.Logging.
    /// Configured runtime creation checks this range only when file output is enabled.
    /// Used by enabled file output as a scheduling target, not a completion or durability guarantee.
    /// </remarks>
    public int FlushIntervalMilliseconds { get; set; } = 1000;

    /// <summary>The ShutdownTimeoutMilliseconds setting.</summary>
    public int ShutdownTimeoutMilliseconds { get; set; } = 5000;

    /// <summary>The FileMaxBytes setting.</summary>
    public long FileMaxBytes { get; set; } = 10485760;

    /// <summary>The FileRetainedFiles setting.</summary>
    public int FileRetainedFiles { get; set; } = 10;

    /// <summary>The SchemaVersion setting.</summary>
    public int SchemaVersion { get; set; } = 1;

    /// <summary>The MaxEntryBytes setting.</summary>
    public int MaxEntryBytes { get; set; } = 65536;

    /// <summary>The console threshold after inheritance.</summary>
    public CoreLogLevel EffectiveConsoleLevel => ConsoleLevel ?? Level;

    /// <summary>The file threshold after inheritance.</summary>
    public CoreLogLevel EffectiveFileLevel => FileLevel ?? Level;

    /// <summary>Whether the console threshold enables output.</summary>
    public bool ConsoleEnabled => EffectiveConsoleLevel != CoreLogLevel.None;

    /// <summary>Whether a path and threshold enable file output.</summary>
    public bool FileEnabled => !string.IsNullOrWhiteSpace(FilePath) && EffectiveFileLevel != CoreLogLevel.None;

    /// <summary>The selected flush interval.</summary>
    public TimeSpan FlushInterval => TimeSpan.FromMilliseconds(FlushIntervalMilliseconds);

    /// <summary>The selected shutdown timeout.</summary>
    public TimeSpan ShutdownTimeout => TimeSpan.FromMilliseconds(ShutdownTimeoutMilliseconds);

    /// <summary>Checks logging settings and normalizes the file path against the application's supplied absolute directory.</summary>
    /// <remarks>Call after assignments and overrides, before starting logging. Performs no output creation or probing.</remarks>
    public void Prepare(string baseDirectory)
    {
        if (string.IsNullOrWhiteSpace(baseDirectory) || !Path.IsPathFullyQualified(baseDirectory))
        {
            throw new ArgumentException("Supply an absolute application startup directory.", nameof(baseDirectory));
        }
        var issues = new List<ConfigurationIssue>();
        CheckEnum(Level, nameof(Level), issues);
        CheckEnum(ConsoleFormat, nameof(ConsoleFormat), issues);
        if (ConsoleLevel is { } consoleLevel)
        {
            CheckEnum(consoleLevel, nameof(ConsoleLevel), issues);
        }
        CheckEnum(ConsoleStream, nameof(ConsoleStream), issues);
        CheckEnum(HumanDetail, nameof(HumanDetail), issues);
        CheckEnum(FileFormat, nameof(FileFormat), issues);
        if (FileLevel is { } fileLevel)
        {
            CheckEnum(fileLevel, nameof(FileLevel), issues);
        }
        CheckRange(QueueCapacity, MinimumQueueCapacity, MaximumQueueCapacity, nameof(QueueCapacity), issues);
        CheckRange(FlushIntervalMilliseconds, MinimumFlushIntervalMilliseconds, MaximumFlushIntervalMilliseconds, nameof(FlushIntervalMilliseconds), issues);
        CheckRange(ShutdownTimeoutMilliseconds, MinimumShutdownTimeoutMilliseconds, MaximumShutdownTimeoutMilliseconds, nameof(ShutdownTimeoutMilliseconds), issues);
        CheckRange(FileMaxBytes, MinimumFileMaxBytes, MaximumFileMaxBytes, nameof(FileMaxBytes), issues);
        CheckRange(FileRetainedFiles, MinimumFileRetainedFiles, MaximumFileRetainedFiles, nameof(FileRetainedFiles), issues);
        CheckRange(MaxEntryBytes, MinimumMaxEntryBytes, MaximumMaxEntryBytes, nameof(MaxEntryBytes), issues);
        if (SchemaVersion != 1)
        {
            issues.Add(new(ConfigurationIssueCodes.UnsupportedSchemaVersion, "LoggingOptions.SchemaVersion", "Only schema version 1 is supported."));
        }
        bool entryAndFileSizesInRange = FileMaxBytes is >= MinimumFileMaxBytes and <= MaximumFileMaxBytes
            && MaxEntryBytes is >= MinimumMaxEntryBytes and <= MaximumMaxEntryBytes;
        if (FileEnabled && entryAndFileSizesInRange && FileMaxBytes < MaxEntryBytes)
        {
            issues.Add(new(ConfigurationIssueCodes.OutOfRange, "LoggingOptions.FileMaxBytes", "An enabled file output requires FileMaxBytes to be at least MaxEntryBytes."));
        }
        string? path = null;
        if (!string.IsNullOrWhiteSpace(FilePath))
        {
            try
            {
                path = LoggingFilePath.Normalize(FilePath.Trim(), baseDirectory);
            }
            catch (Exception error) when (error is ArgumentException or NotSupportedException or PathTooLongException)
            {
                issues.Add(new(ConfigurationIssueCodes.InvalidPath, "LoggingOptions.FilePath", "Supply a syntactically valid file path."));
            }
        }
        if (issues.Count != 0)
        {
            throw new CoreConfigurationException(issues);
        }
        FilePath = path;
    }

    private static void CheckEnum<T>(T value, string name, List<ConfigurationIssue> issues) where T : struct, Enum
    {
        if (!Enum.IsDefined(value))
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidToken, "LoggingOptions." + name, "Use a declared option value."));
        }
    }

    private static void CheckRange(long value, long minimum, long maximum, string name, List<ConfigurationIssue> issues)
    {
        if (value < minimum || value > maximum)
        {
            issues.Add(new(ConfigurationIssueCodes.OutOfRange, "LoggingOptions." + name, "The value is outside the supported range."));
        }
    }
}

using System.ComponentModel;
using System.Diagnostics;
using System.Reflection;
using System.Runtime.InteropServices;
using CIS.Configuration;
using CIS.Logging;
using CIS.Text;

namespace CIS;

/// <summary>Shared application and runtime facts populated automatically on first access.</summary>
/// <remarks>A static constructor applies discovered defaults followed by known environment overrides. It starts no module runtime.</remarks>
public static class EnvironmentOptions
{
    static EnvironmentOptions()
    {
        Assembly? application = Assembly.GetEntryAssembly();
        BuildVersion = ReadBuildVersion(application);
        ApplicationName = application?.GetName().Name ?? "unknown";
        EnvironmentName = "unknown";
        ServiceVersion = BuildVersion;
        HostName = ReadHostName();
        OperatingSystem = ReadOperatingSystemFamily();
        OperatingSystemDescription = RuntimeInformation.OSDescription;
        OSArchitecture = RuntimeInformation.OSArchitecture.ToString().ToLowerInvariant();
        ProcessArchitecture = RuntimeInformation.ProcessArchitecture.ToString().ToLowerInvariant();
        RuntimeDescription = RuntimeInformation.FrameworkDescription;
        ProcessId = System.Environment.ProcessId;
        ProcessStartedAtUtc = ReadProcessStartedAtUtc();
        InstanceId = Guid.NewGuid().ToString("D");
        InitializedAtUtc = DateTimeOffset.UtcNow;
        Logging = new LoggingOptions();
        var issues = new List<ConfigurationIssue>();
        (ApplicationName, EnvironmentName, ServiceVersion) = EnvironmentOverrides.Apply(
            System.Environment.GetEnvironmentVariable, ApplicationName, EnvironmentName, ServiceVersion, Logging, issues);
        if (issues.Count != 0) { throw new CoreConfigurationException(issues); }
        CheckIdentity(ApplicationName, nameof(ApplicationName), 255, issues);
        CheckIdentity(EnvironmentName, nameof(EnvironmentName), 255, issues);
        CheckIdentity(BuildVersion, nameof(BuildVersion), 512, issues);
        CheckIdentity(ServiceVersion, nameof(ServiceVersion), 512, issues);
        if (issues.Count != 0) { throw new CoreConfigurationException(issues); }
        // Intern only validated, bounded identities after all overrides. Never intern UUIDs, paths or payloads.
        ApplicationName = string.Intern(ApplicationName);
        EnvironmentName = string.Intern(EnvironmentName);
        BuildVersion = string.Intern(BuildVersion);
        ServiceVersion = string.Intern(ServiceVersion);
    }

    /// <summary>The selected application name; interned after overrides.</summary>

    /// <summary>The actual application build version; independent of service-version overrides and interned.</summary>
    public static string BuildVersion { get; }

    public static string ServiceVersion { get; }
    /// <summary>The selected deployment environment label; interned after overrides.</summary>
    public static string EnvironmentName { get; }
    /// <summary>The application-selected execution kind; defaults to Unknown.</summary>
    public static EnvironmentKind EnvironmentKind { get; set; }
    /// <summary>The process-wide application instance UUID, without interning.</summary>
    public static string InstanceId { get; }
    /// <summary>The process start instant in UTC, or null when host inspection is unavailable.</summary>
    public static DateTimeOffset? ProcessStartedAtUtc { get; }
    /// <summary>The UTC instant at which the static container was populated.</summary>
    public static DateTimeOffset InitializedAtUtc { get; }
    /// <summary>The current host name.</summary>
    public static string? HostName { get; }
    /// <summary>The lowercase operating-system family.</summary>
    public static string OperatingSystem { get; }
    /// <summary>The observed operating-system description.</summary>
    public static string? OperatingSystemDescription { get; }
    /// <summary>The lowercase operating-system architecture.</summary>
    public static string? OSArchitecture { get; }
    /// <summary>The lowercase process architecture.</summary>
    public static string? ProcessArchitecture { get; }
    /// <summary>The observed runtime description.</summary>

    /// <summary>The current positive process identifier.</summary>
    public static int? ProcessId { get; }
    /// <summary>The optional caller-supplied build commit identifier.</summary>

    /// <summary>The optional caller-supplied build identifier.</summary>

    /// <summary>Mutable module settings; logging owns their preparation and runtime.</summary>
    public static LoggingOptions Logging { get; }

    internal static string ReadBuildVersion(Assembly? application)
    {
        string? information = application?.GetCustomAttribute<AssemblyInformationalVersionAttribute>()?.InformationalVersion;
        return string.IsNullOrWhiteSpace(information) ? application?.GetName().Version?.ToString() ?? "unknown" : information.Trim();
    }

    private static void CheckIdentity(string value, string name, int maximum, List<ConfigurationIssue> issues)
    {
        if (!UnicodeScalarText.IsValid(value, maximum, allowControls: false))
        {
            issues.Add(new(ConfigurationIssueCodes.InvalidIdentity, "EnvironmentOptions." + name, "Supply a bounded identity without controls or malformed Unicode."));
        }
    }

    private static string ReadOperatingSystemFamily()
    {
        if (System.OperatingSystem.IsWindows()) { return "windows"; }
        if (System.OperatingSystem.IsLinux()) { return "linux"; }

        private long ticks;
        private DateTimeOffset utc = new(2026, 10, 13, 12, 0, 0, TimeSpan.Zero);

        public override long GetTimestamp() { lock (gate) { return ticks; } }
        public override DateTimeOffset GetUtcNow() { lock (gate) { return utc; } }

        {

            if (dueTime != Timeout.InfiniteTimeSpan || period != Timeout.InfiniteTimeSpan) { timer.Change(dueTime, period); }
            return timer;
        }
        internal void Advance(TimeSpan elapsed)
        {
            Require(elapsed >= TimeSpan.Zero);
            long target; lock (gate) { target = checked(ticks + elapsed.Ticks); }
            while (true)
            {
                Timer[] ready;
                lock (gate)
                {
                    long next = timers.Select(t => t.Due).DefaultIfEmpty(long.MaxValue).Min();
                    long destination = Math.Min(next, target);
                    utc += TimeSpan.FromTicks(destination - ticks); ticks = destination;
                    ready = timers.Where(t => t.Due <= ticks).ToArray();
                    foreach (Timer timer in ready)
                    {
                        if (timer.Period > 0) { timer.Due = checked(timer.Due + timer.Period); PeriodicFires++; }
                        else { timer.Due = long.MaxValue; DeadlineFires++; }
                    }
                    if (ready.Length == 0) { return; }
                }
                foreach (Timer timer in ready) { timer.Callback(timer.State); }
            }
        }
        internal JsonObject Report()
        {
            lock (gate)
            {
                return new JsonObject
                {
                    ["created"] = Created,
                    ["disposed"] = Disposed,
                    ["periodicFires"] = PeriodicFires,
                    ["deadlineFires"] = DeadlineFires,
                    ["active"] = timers.Count,
                    ["activePeriodic"] = timers.Count(t => t.Period > 0),
                    ["maximumActive"] = MaximumActive,
                    ["periodicSuppressed"] = PeriodicSuppressed,
                    ["systemCreated"] = SystemCreated,
                    ["systemFalseChanges"] = SystemFalseChanges
                };
            }
        }
        private sealed class Timer(Clock clock, TimerCallback callback, object? state, bool suppressed, ITimer? actual) : ITimer
        {
            internal readonly TimerCallback Callback = callback; internal readonly object? State = state;
            internal long Due = long.MaxValue, Period; private bool disposed;
            public bool Change(TimeSpan dueTime, TimeSpan period)
            {
                if (clock.HoldArm && period > TimeSpan.Zero) { clock.ArmEntered.TrySetResult(); Require(clock.ArmRelease.Task.Wait(Guard)); }
                lock (clock.gate)
                {
                    if (actual is not null)
                    {
                        bool changed = actual.Change(dueTime, period);
                        if (!changed) { clock.SystemFalseChanges++; }
                        return changed;
                    }
                    if (!disposed && clock.ThrowOwned && period > TimeSpan.Zero) { throw new ObjectDisposedException("controlled timer"); }
                    if (disposed || clock.ReturnFalse && period > TimeSpan.Zero) { return false; }
                    Period = period == Timeout.InfiniteTimeSpan ? 0 : period.Ticks;
                    Due = dueTime == Timeout.InfiniteTimeSpan ? long.MaxValue : checked(clock.ticks + dueTime.Ticks);
                    if (Period > 0) { clock.PeriodicSuppressed &= suppressed; }
                    return true;
                }
            }
            public void Dispose()
            {
                lock (clock.gate) { if (!disposed) { disposed = true; clock.timers.Remove(this); clock.Disposed++; } }
                actual?.Dispose();
            }
            public ValueTask DisposeAsync() { Dispose(); return actual?.DisposeAsync() ?? ValueTask.CompletedTask; }
        }
