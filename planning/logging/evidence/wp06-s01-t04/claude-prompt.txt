using System.Collections;
using System.Globalization;
using CIS.Logging.Values;

namespace CIS.Logging.MicrosoftLogger;

// A single bounded synchronous copy; no template parsing or arbitrary object serialization.
internal sealed class ProviderStateCapture
{
    private int remaining = LogValueProjection.MaximumNodes - 2; // Root plus provider-owned event_id.
    private readonly HashSet<object> ancestors = new(ReferenceEqualityComparer.Instance);

    internal static (string? Template, LogValue Attributes) Capture<TState>(TState state, int eventId)
    {
        var capture = new ProviderStateCapture();
        var attributes = new Dictionary<string, LogValue>(StringComparer.Ordinal) { ["event_id"] = LogValue.Integer(eventId) };
        if (state is not IEnumerable<KeyValuePair<string, object?>> fields)
        { return (null, LogValue.OwnedMap(attributes, truncated: false)); }
        string? template = null; bool truncated = false; int visited = 0;
        capture.Enter(fields);
        try
        {
            foreach ((string key, object? value) in fields)
            {
                if (++visited > LogValueProjection.MaximumNodes) { truncated = true; break; }
                if (key == "{OriginalFormat}")
                {
                    if (value is not null and not string) { throw new ArgumentException("OriginalFormat must be a string or null."); }
                    template = (string?)value; continue;
                }
                if (key == "event_id") { continue; }
                // Keep scanning bounded root metadata after the value budget is spent.
                if (capture.remaining == 0) { truncated = true; continue; }
                ValidateKey(key);
                LogValue? retained = capture.Value(value, depth: 2);
                if (retained is null) { truncated = true; continue; }
                attributes[key] = retained;
            }
        }
        finally { capture.ancestors.Remove(fields); }
        return (template, LogValue.OwnedMap(attributes, truncated));
    }

    private LogValue? Value(object? value, int depth)
    {
        if (remaining == 0 || depth > LogValueProjection.MaximumDepth) { return null; }
        if (value is LogValue existing && existing.NodeCount <= remaining && depth - 1 + existing.Depth <= LogValueProjection.MaximumDepth)
        { remaining -= existing.NodeCount; return existing; }
        remaining--;
        return value switch
        {
            null => LogValue.Null(),
            LogValue owned when owned.Kind == LogValueKind.Array => Array(owned.ArrayItems, depth, owned.IsTruncated),
            LogValue owned when owned.Kind == LogValueKind.Map => Map(owned.MapProperties.Select(p => new KeyValuePair<string, object?>(p.Key, p.Value)), depth, owned.IsTruncated),
            bool scalar => LogValue.Boolean(scalar),
            string scalar => LogValue.String(scalar),
            char scalar => LogValue.String(new string(scalar, 1)),
            sbyte scalar => LogValue.Integer(scalar),
            byte scalar => LogValue.Integer(scalar),
            short scalar => LogValue.Integer(scalar),
            ushort scalar => LogValue.Integer(scalar),
            int scalar => LogValue.Integer(scalar),
            uint scalar => LogValue.Integer(scalar),
            long scalar => LogValue.Integer(scalar),
            ulong scalar when scalar <= 9_007_199_254_740_991 => LogValue.Integer((long)scalar),
            float scalar => LogValue.Number(scalar),
            double scalar => LogValue.Number(scalar),
            decimal scalar => Decimal(scalar),
            Guid scalar => LogValue.Uuid(scalar),
            DateTimeOffset scalar => LogValue.Timestamp(scalar),
            DateTime scalar when scalar.Kind == DateTimeKind.Utc => LogValue.Timestamp(new DateTimeOffset(scalar)),
            IEnumerable<KeyValuePair<string, object?>> properties => Map(properties, depth),
            IDictionary properties => Map(Entries(properties), depth, owner: properties),
            IEnumerable items => Array(items, depth),
            _ => throw new ArgumentException("Use supported scalar, collection or explicit LogValue inputs.")
        };
    }

    private LogValue Array(IEnumerable items, int depth, bool truncated = false)
    {
        Enter(items);
        try
        {
            var owned = new List<LogValue>();
            foreach (object? item in items)
            {
                LogValue? retained = Value(item, depth + 1);
                if (retained is null) { truncated = true; break; }
                owned.Add(retained);
            }
            return LogValue.OwnedArray(owned, truncated);
        }
        finally { ancestors.Remove(items); }
    }

    private LogValue Map(IEnumerable<KeyValuePair<string, object?>> properties, int depth, bool truncated = false, object? owner = null)
    {
        owner ??= properties; Enter(owner);
        try
        {
            var owned = new Dictionary<string, LogValue>(StringComparer.Ordinal);
            foreach ((string key, object? value) in properties)
            {
                ValidateKey(key);
                LogValue? retained = Value(value, depth + 1);
                if (retained is null) { truncated = true; break; }
                owned[key] = retained;
            }
            return LogValue.OwnedMap(owned, truncated);
        }
        finally { ancestors.Remove(owner); }
    }

    private static IEnumerable<KeyValuePair<string, object?>> Entries(IDictionary properties)
    {
        foreach (DictionaryEntry entry in properties)
        { yield return new(entry.Key as string ?? throw new ArgumentException("Map keys must be strings."), entry.Value); }
    }

    private void Enter(object collection)
    {
        if (!ancestors.Add(collection)) { throw new ArgumentException("Cyclic state is unsupported."); }
    }
    private static void ValidateKey(string key)
    {
        if (key is null || !LogValueText.IsValidKey(key)) { throw new ArgumentException("Use valid ordinal attribute keys."); }
    }
    private static LogValue Decimal(decimal value)
    {
        string text = value.ToString(CultureInfo.InvariantCulture);
        return LogValue.DecimalString(text.Contains('.', StringComparison.Ordinal) ? text : text + ".0");
    }
}

using Microsoft.Extensions.Logging;

namespace CIS.Logging.MicrosoftLogger;

/// <summary>Creates Microsoft category loggers that borrow an application-owned core runtime.</summary>
/// <remarks>Disposal disables this provider's future calls and never stops or disposes the shared runtime.
/// A call already running when disposal occurs may finish. Category names follow the core component rules.
/// Supplied rendered messages and exceptions enter core capture. Null or empty EventId.Name becomes
/// log.message; EventId.Id supplies the integer attribute event_id, including zero and negative values.
/// Structured state preserves {OriginalFormat} without parsing. Core checks interest before invoking
/// formatter or copying state; full queues remain interested. Formatter exceptions propagate unchanged;
/// invalid retained payload increments core diagnostics without output admission. Oversized valid payload
/// is truncated. Disposed, stopped, filtered and all-faulted calls skip both callbacks after argument guards.
/// Healthy outputs admit independently; host scopes remain integration work.</remarks>
public sealed class CoreLoggerProvider : ILoggerProvider
{
    private readonly CoreLogRuntime runtime;
    private int disposed;

    /// <summary>Creates a nonowning provider over an existing runtime.</summary>
    /// <exception cref="ArgumentNullException">Runtime is null.</exception>
    public CoreLoggerProvider(CoreLogRuntime runtime)
    {
        ArgumentNullException.ThrowIfNull(runtime);
        this.runtime = runtime;
    }

    /// <summary>Creates a category logger using the core's validated component name.</summary>
    /// <exception cref="ObjectDisposedException">This provider has been disposed.</exception>
    /// <exception cref="ArgumentNullException">Category name is null.</exception>
    /// <exception cref="ArgumentException">The name does not satisfy the core component rules.</exception>
    public ILogger CreateLogger(string categoryName)
    {
        ObjectDisposedException.ThrowIf(IsDisposed, this);
        return new CategoryLogger(this, runtime.CreateLogger(categoryName));
    }

    private bool IsDisposed => Volatile.Read(ref disposed) != 0;

    /// <summary>Idempotently disables this provider; the application retains runtime ownership.</summary>
    public void Dispose() => Interlocked.Exchange(ref disposed, 1);

    private static CoreLogLevel Level(LogLevel logLevel)
    {
        if (!Enum.IsDefined(logLevel)) { throw new ArgumentOutOfRangeException(nameof(logLevel)); }
        // The selected v1 Core and Microsoft severity numbers are identical (Trace through None).
        return (CoreLogLevel)logLevel;
    }

    private sealed class CategoryLogger(CoreLoggerProvider provider, CoreLogger core) : ILogger
    {
        public bool IsEnabled(LogLevel logLevel)
        {
            CoreLogLevel level = Level(logLevel);
            return !provider.IsDisposed && core.IsEnabled(level);
        }

        public IDisposable? BeginScope<TState>(TState state) where TState : notnull
        {
            ArgumentNullException.ThrowIfNull(state);
            ObjectDisposedException.ThrowIf(provider.IsDisposed, provider);
            if (state is not OperationContextData context)
            { throw new ArgumentException("Use OperationContextData for the provider foundation's typed core scope.", nameof(state)); }
            if (provider.runtime.GetDiagnostics().AdmissionClosed) { throw new InvalidOperationException("Cannot begin a scope after runtime admission closes."); }
            return provider.runtime.Scopes.BeginScope(context);
        }

        public void Log<TState>(LogLevel logLevel, EventId eventId, TState state, Exception? exception,
            Func<TState, Exception?, string> formatter)
        {
            ArgumentNullException.ThrowIfNull(formatter);
            CoreLogLevel level = Level(logLevel);
            if (provider.IsDisposed) { return; }
            _ = core.WriteLazy(level, () => new LogEventData
            {
                EventName = string.IsNullOrEmpty(eventId.Name) ? "log.message" : eventId.Name,
                Message = formatter(state, exception),
                Exception = exception,
                CaptureState = () => ProviderStateCapture.Capture(state, eventId.Id)
            });
        }
    }
}

# Microsoft structured state

The provider keeps the supplied formatter as the only message formatter. It passes the original state and exception to that formatter after Core's interest check. State implementing IEnumerable<KeyValuePair<string, object?>> is copied as ordinal attributes. Other state remains formatter input only. The exact {OriginalFormat} key supplies message_template when its value is a string; null means absent. It is excluded from attributes. Core clips template text at its existing scalar and aggregate entry bounds; the provider does not parse it.

Normalization runs synchronously inside Core capture after Activity IDs and the event clock have been captured. The internal LogEventData capture callback returns an owned template/map and never reaches an output queue. Caller collection/normalization failures become Core InvalidEvent accounting, without partial admission. Supplied formatter exceptions still propagate unchanged outside that capture catch. This private seam adds no public member, worker, dependency or general serialization framework.

Supported values are null, Boolean, string/char, SByte, Byte, Int16, UInt16, Int32, UInt32, Int64 and UInt64 within the portable safe-integer range, finite Single/Double, Decimal, Guid, DateTimeOffset, UTC DateTime, and explicit immutable LogValue values. Integral magnitudes never change a number to a string; use LogValue.IntegerString explicitly for larger integers. Decimal uses invariant decimal-string form, including .0 for whole values. Timestamps are converted to UTC through the existing value factory; non-UTC DateTime is rejected instead of guessing an offset. Single is widened exactly to Double: 0.1f becomes 0.10000000149011612 in the numeric attribute even if the formatter displays 0.1. Enums, TimeSpan, Uri, IPAddress, native integers, Int128/UInt128, BigInteger, arbitrary objects and nonfinite numbers are rejected. No arbitrary ToString or reflection serialization is used.

Nested arrays/enumerables and maps are copied recursively. Nested maps accept string-key IDictionary or IEnumerable<KeyValuePair<string, object?>> forms. A root Dictionary<string, int> does not implement the supported root state shape and remains formatter input only; it can be copied as a nested map. Keys keep their ordinal casing and follow Core's Unicode/control/length rules. Last visited duplicate keys win. The provider supplies event_id before copying and ignores any state value for that key, so state or typed scope cannot replace its numeric input. Ordinary attribute projection still follows Core's aggregate truncation rules; numeric ID is not a new protected envelope field.

One bounded copy budget includes the attributes root and provider event_id: at most 1024 nodes and depth 8. Nested copying stops at the first excess child and carries truncation into Core's merged event. After the ordinary value budget is exhausted, the root scan continues only to recognize {OriginalFormat}; other keys/values are skipped without validation or copying. Replaced duplicate values count toward copy work; root state enumeration also has a 1024-entry guard so repeated metadata/ignored keys cannot cause an unbounded secondary copy. Unvisited tails, including templates beyond that prefix, are omitted with truncation. Collections remain cooperative caller code; this does not impose an execution deadline on arbitrary enumerator callbacks. Cycles observed in the retained traversal are invalid. No caller map/array remains mutable in queued facts; already immutable LogValue subtrees can be shared.

Nine cases exercise actual Microsoft extension formatting, every supported scalar family, copied nested collections, duplicate/event/scope precedence, unsupported/cyclic input, node/depth/traversal bounds, skipped formatter/state traversal, Activity mutation during enumeration and template clipping. Physical variants use public runtime creation, real JSONL bytes and an exclusive lease probe after stop. The physical borrowedConsoleOpen field is null because that variant does not select a console. Executed reports and independent native-reader controls belong in the task receipt.

[Interest and rejection](microsoft-provider-behavior.md) qualify T04 behavior, including standard Microsoft template-last ordering. T05 still qualifies the complete story. S02 retains host/external scopes, final provider surface and host lifetime integration; examples, instructions/skills and release remain required. Original plans and historical receipts are preserved.

# Microsoft provider interest and rejection

ILogger.IsEnabled delegates to Core after validating the level. None, no interested output, closed admission, a disposed provider, and all interested outputs faulted return false. A full queue remains interested: Log still formats and captures once, then Core counts each full output without admitting a ticket. Each healthy interested output admits independently; a faulted output does not prevent another output from accepting the same capture. There is no automatic writer retry or replacement.

Log validates the formatter before the level, even after disposal or shutdown. Before either callback, Core applies closure/interest/fault gates. Filtered calls update existing output filtered counts; all-faulted interested calls update unavailable counts; stopped and disposed calls perform no capture. Provider disposal disables its loggers and leaves application runtime ownership intact. Concurrent disposal/closure after the first check follows the existing Core capture/admission race contract.

The supplied formatter receives the original state and exception once. Its exception propagates unchanged without invalid-event accounting. A null or invalid Unicode message, invalid event name, unsupported retained state, malformed OriginalFormat, or cooperative state enumeration failure counts one Core InvalidEvent and admits nothing to any output. ILogger.Log is void and discards Core's result; applications observe loss through Core diagnostics and flush/stop results. Capture callback failures use the existing fixed UnsupportedValue category; malformed adapted templates currently use this same category because the template is supplied inside capture. No exception text becomes a fault category. Argument guards still throw. Owned encoder consistency failures and catastrophic failures keep their Core behavior.

Oversized valid payloads are truncated under existing node, depth, scalar and entry-byte bounds. They count as truncated rather than invalid. Root metadata scanning continues under the 1024-entry limit after ordinary value copying stops, preserving Microsoft's final OriginalFormat when visited. Unsupported keys/values beyond the retained budget are omitted, not validated. A template beyond the entry scan guard is also omitted with truncation. EventId is ordinary payload under merged scope and byte truncation; aggregate projection can omit it. Duplicate OriginalFormat and nested keys use the last visited value. Supported type/root shape details are in [structured state](microsoft-provider-state.md).

State enumeration is synchronous cooperative caller code. The formatter can itself enumerate a collection before capture; single-use enumerables can therefore fail the second traversal and invalidate the whole event. Materialize them before logging. There is no imposed deadline, arbitrary ToString fallback or reflection serializer.

Fourteen fixture modes and argument-guard cases qualify these decisions. Public physical consumers cover filters, stopped/disposed calls, template ordering, float widening, duplicates/root shapes, unsupported input, capture callbacks and formatter exceptions. No-output, queue-full and writer-fault cases use actual internal Core queues with controlled callbacks, explicit barriers and memory bytes; they do not claim an induced operating-system failure. Independent native readers bind child PID, exact assemblies, raw output bytes and corruption controls. Execution and review results belong in the task receipt.

T05 complete story acceptance, S02 host/external scopes and lifetime integration, examples/skills and release qualification remain pending. No public API, dependency or worker was added in T04.

    private static CoreLogRuntime ComposeQueued(EventMetadata metadata, LoggingOptions selected,
        Func<AdmittedEvent, ValueTask>? consoleWrite, Func<AdmittedEvent, ValueTask>? fileWrite,
        TimeProvider? timeProvider, ConsoleOutputWriter? writer = null, Func<Func<Task>, Task>? launchWorker = null, FileOutputWriter? fileWriter = null, Func<ValueTask>? consoleFlush = null, Func<ValueTask>? fileFlush = null)

        ValidateSettings(selected.EffectiveConsoleLevel, selected.EffectiveFileLevel, selected.MaxEntryBytes);
        var gate = new object();
        var queues = new OutputQueues(selected.QueueCapacity, selected.ConsoleEnabled, selected.FileEnabled, consoleWrite, fileWrite, gate, launchWorker, fileWriter,
            writer is null ? consoleFlush : writer.FlushAsync, fileWriter is null ? fileFlush : fileWriter.FlushAsync,
            writer is null ? null : writer.WriteAsync, fileWriter is null ? null : fileWriter.WriteAsync,
            writer is null ? null : writer.FlushAsync, fileWriter is null ? null : fileWriter.FlushAsync);
        try
        {
            writer?.Initialize();
            fileWriter?.Initialize();
            var runtime = new CoreLogRuntime(metadata, selected, queues.Console, queues.File, timeProvider, gate, queues, writer, fileWriter);
            if (fileWriter is not null)
            {
                try { queues.File!.StartFileFlushTimer(runtime.timeProvider, selected.FlushInterval); }
                catch { throw new CoreLogInitializationException("file", "FileFlushTimerInitializationFailed"); }
            }
            return runtime;
        }
        catch
        {
            queues.Complete();
            // Observe terminal cleanup failures without masking the safe initialization category.
            try { queues.Completion.GetAwaiter().GetResult(); } catch { }
            try { fileWriter?.Dispose(); } catch { }
            throw;
        }

    private static void ValidateShutdownTimeout(int milliseconds)

    /// <summary>Starts or joins one shared bounded stop operation through its original accepted boundaries.</summary>
    /// <param name="cancellationToken">Cancels only this caller's wait; even an already-cancelled first caller starts shutdown.</param>
    /// <returns>The same owned immutable observation retained for all joining callers.</returns>
    /// <exception cref="OperationCanceledException">The caller cancels its own wait.</exception>
    /// <remarks>The configured overall timeout is frozen at creation. UTC is an observation label; elapsed time is monotonic. At expiry, unstarted accepted records are abandoned. Cancellation of active writes/flushes is requested but is not guaranteed, and no new flush, retry or replacement writer starts. Completed means terminal dispositions, not successful delivery or resource release. A timed-out observation can retain pending I/O; its later completion does not change the cached result.</remarks>
    public Task<LogStopResult> StopAsync(CancellationToken cancellationToken)
    {
        if (queuedOutputs is null) { throw new InvalidOperationException("Stop requires actual queued output composition."); }
        Task<LogStopResult>? shared;
        lock (admissionGate) { shared = stopTask; }
        if (shared is null)
        {
            // Caller clock code is never invoked under admission synchronization.
            // Preliminary concurrent samples are harmless; only the winning caller owns the deadline.
            long started = timeProvider.GetTimestamp();
            RuntimeFlushBoundary boundary = default;
            TaskCompletionSource<LogStopResult>? completion = null;
            lock (admissionGate)
            {
                if (stopTask is null)
                {
                    completion = new(TaskCreationOptions.RunContinuationsAsynchronously);
                    stopTask = completion.Task;
                    try
                    {
                        // Register final controls before a closing idle worker is allowed to exit.
                        boundary = RegisterFlushBoundary();
                        shutdownStartedAtUtc ??= boundary.BoundaryCapturedAtUtc;
                        admissionClosed = true;
                        queuedOutputs.Complete();
                    }
                    catch (Exception error)
                    {
                        // A failed setup must not leave the already-published task pending forever.
                        completion.TrySetException(error);
                        throw;
                    }
                }
                shared = stopTask;
            }
            if (completion is not null)
            {
                // Start deadline acquisition directly, without a delayed ThreadPool launch.
                // Only continuations/timers outlive this call; they do not retain the caller's flow.
                if (ExecutionContext.IsFlowSuppressed()) { _ = RunStopAsync(boundary, started, completion); }
                else { using (ExecutionContext.SuppressFlow()) { _ = RunStopAsync(boundary, started, completion); } }
            }
        }
        // WaitAsync can prefer an already-completed task; an already-cancelled caller must still cancel.
        if (cancellationToken.IsCancellationRequested) { return Task.FromCanceled<LogStopResult>(cancellationToken); }
        return cancellationToken.CanBeCanceled ? shared!.WaitAsync(cancellationToken) : shared!;
    }

    /// <summary>Starts or joins the original bounded stop operation without a new drain window.</summary>
    /// <returns>A wait on the same shared operation; unsuccessful draining remains visible in LastStopResult.</returns>
    /// <remarks>Repeated disposal is harmless. At the original deadline this wait may complete while actual I/O and owned cleanup remain pending; no resource is disposed beneath active I/O. Inspect GetDiagnostics().LastStopResult for the immutable drain outcome.</remarks>
    public ValueTask DisposeAsync() => new(StopAsync(CancellationToken.None));

    private async Task RunStopAsync(RuntimeFlushBoundary boundary, long started, TaskCompletionSource<LogStopResult> completion)
    {
        try
        {
            if (boundary.Console is null && boundary.File is null)
            { PublishStop(CaptureFlushResult(boundary), completion); return; }
            await using var deadline = new FlushDeadline(timeProvider, started, shutdownTimeout, () => CaptureFlushResult(boundary), () => CaptureStopDeadline(boundary));
            try
            {
                await boundary.WaitForWriterFlushAsync(deadline.Token).ConfigureAwait(false);
                if (boundary.Console is { } console)
                { await queuedOutputs!.Console!.WaitForCoveredRecordsAsync("console", console, deadline.Token).ConfigureAwait(false); }
                if (boundary.File is { } file)
                { await queuedOutputs!.File!.WaitForCoveredRecordsAsync("file", file, deadline.Token).ConfigureAwait(false); }
                try { await queuedOutputs!.Completion.WaitAsync(deadline.Token).ConfigureAwait(false); }
                catch (Exception) when ((queuedOutputs!.Completion.IsFaulted || queuedOutputs.Completion.IsCanceled) && (queuedOutputs.Console?.IsFaulted == true || queuedOutputs.File?.IsFaulted == true))
                { /* Terminal writer faults remain safe output results, not business-call I/O exceptions. */ }
            }
            catch (OperationCanceledException) when (deadline.Token.IsCancellationRequested) { }
            // Publish the frozen result before asynchronous deadline-timer cleanup can wait on a callback join.
            PublishStop(deadline.ObserveCompletion(), completion);
        }
        catch (Exception error)
        {
            if (completion.Task.IsCompletedSuccessfully)
            {
                // Publication precedes deadline-timer cleanup; later failure belongs to live diagnostics.
                queuedOutputs!.RecordCleanupFault("ShutdownTimerCleanupFailed");
            }
            else { completion.TrySetException(error); }
        }
    }

    private LogFlushResult CaptureStopDeadline(RuntimeFlushBoundary boundary)
    {
        LogFlushResult observation;
        lock (admissionGate)
        {
            queuedOutputs!.AbandonUnstarted();
            observation = CaptureFlushResult(boundary);
        }
        // Freeze actual pending work before cooperative cancellation can settle it.
        queuedOutputs.RequestIoCancellation();
        return observation;
    }
    private void PublishStop(LogFlushResult observation, TaskCompletionSource<LogStopResult> completion)
    {
        var result = new LogStopResult(observation);
        lock (admissionGate) { lastStopResult = result; }
        completion.TrySetResult(result);
    }


    /// <summary>Waits through fixed accepted boundaries and confirms each enabled writer's buffer flush.</summary>
    /// <param name="timeout">One positive finite overall wait budget; every positive TimeSpan is supported.</param>
    /// <param name="cancellationToken">Cancels only this caller's wait, preserving shared worker requests.</param>
    /// <returns>An owned immutable observation, including unresolved work at the deadline.</returns>
    /// <exception cref="ArgumentOutOfRangeException">The timeout is zero, negative or infinite.</exception>
    /// <exception cref="OperationCanceledException">The caller token is cancelled, including after boundary registration.</exception>
    /// <remarks>UTC labels are system observations; elapsed deadlines use the runtime TimeProvider. Flush does not stop admission or guarantee disk durability. After closure it cannot reopen a writer; a terminal writer returns an unavailable/failed flush even if the original stop flushed successfully. Inspect LastStopResult for that original shutdown outcome.</remarks>
    public Task<LogFlushResult> FlushAsync(TimeSpan timeout, CancellationToken cancellationToken)
    {
        if (timeout.Ticks <= 0) { throw new ArgumentOutOfRangeException(nameof(timeout)); }
        long started = timeProvider.GetTimestamp();
        RuntimeFlushBoundary boundary = RegisterFlushBoundary();
        return ObserveFlushAsync(boundary, started, timeout, cancellationToken);
    }
    private async Task<LogFlushResult> ObserveFlushAsync(RuntimeFlushBoundary boundary, long started, TimeSpan timeout, CancellationToken caller)
    {
        caller.ThrowIfCancellationRequested();
        if (boundary.Console is null && boundary.File is null) { return CaptureFlushResult(boundary); }
        await using var deadline = new FlushDeadline(timeProvider, started, timeout, () => CaptureFlushResult(boundary));
        using var wait = CancellationTokenSource.CreateLinkedTokenSource(caller, deadline.Token);
        try
        {
            await boundary.WaitForWriterFlushAsync(wait.Token).ConfigureAwait(false);
            // A terminal unavailable flush can precede the disposition of an already-started write.
            if (boundary.Console is { } console)
            { await queuedOutputs!.Console!.WaitForCoveredRecordsAsync("console", console, wait.Token).ConfigureAwait(false); }
            if (boundary.File is { } file)
            { await queuedOutputs!.File!.WaitForCoveredRecordsAsync("file", file, wait.Token).ConfigureAwait(false); }
        }
        catch (OperationCanceledException) when (wait.IsCancellationRequested)
        { caller.ThrowIfCancellationRequested(); }
        caller.ThrowIfCancellationRequested();
        return deadline.ObserveCompletion();
    }
    private LogFlushResult CaptureFlushResult(RuntimeFlushBoundary boundary)
    {
        lock (admissionGate)
        {
            var observations = new List<LogOutputDrainResult>(2);
            if (boundary.Console is { } console) { observations.Add(queuedOutputs!.Console!.CaptureDrain("console", console)); }
            if (boundary.File is { } file) { observations.Add(queuedOutputs!.File!.CaptureDrain("file", file)); }
            return new LogFlushResult(boundary.BoundaryCapturedAtUtc, DateTimeOffset.UtcNow,
                boundary.InvalidEventCount, boundary.TruncatedEventCount, observations);
        }
    }
    internal RuntimeFlushBoundary RegisterFlushBoundary()
    {
        if (queuedOutputs is null) { throw new InvalidOperationException("Flush boundaries require actual queued output composition."); }
        lock (admissionGate)
        {
            // Pure system UTC sampling here does not invoke caller factories, payloads or the event clock.
            DateTimeOffset capturedAt = DateTimeOffset.UtcNow;
            OutputFlushBoundary? console = queuedOutputs.Console?.RegisterFlushBoundary();
            OutputFlushBoundary? file = queuedOutputs.File?.RegisterFlushBoundary();
            return new RuntimeFlushBoundary(capturedAt, InvalidEventCount, TruncatedEventCount, console, file);
        }
    }
    internal RuntimeAccounting CaptureAccounting()
    {
        if (queuedOutputs is null) { throw new InvalidOperationException("Accounting requires actual queued output composition."); }
        lock (admissionGate)
        {
            // Output/disposition/truncation/closure changes share this gate. Invalid increments
            // independently, so its atomic read chooses the observation point while the rest is stable.
            return new RuntimeAccounting(admissionClosed, InvalidEventCount, TruncatedEventCount,
                queuedOutputs.Console?.CaptureAccounting(), queuedOutputs.File?.CaptureAccounting());
        }
    }

    /// <summary>Returns an owned immutable live observation without routing a log or performing output I/O.</summary>
    /// <remarks>Available after faults and admission closure. This does not register a fixed flush target.</remarks>
    public CoreLogDiagnostics GetDiagnostics()
    {
        lock (admissionGate)
        {
            RuntimeAccounting counts = CaptureAccounting();
            var values = new List<LogOutputDiagnostics>(2);
            if (counts.Console is OutputAccounting console) { values.Add(new LogOutputDiagnostics("console", console)); }
            if (counts.File is OutputAccounting file) { values.Add(new LogOutputDiagnostics("file", file, recoveredPartialSegment)); }
            CoreLogState state = !counts.AdmissionClosed ? CoreLogState.Running :
                values.All(value => value.State == LogOutputState.Stopped) ? CoreLogState.Stopped : CoreLogState.Stopping;
            return new CoreLogDiagnostics(state, counts.AdmissionClosed, DateTimeOffset.UtcNow, shutdownStartedAtUtc,
                counts.InvalidEventCount, counts.TruncatedEventCount, values, lastStopResult);
        }
    }

    internal OperationScopeStore Scopes { get; } = new();
    internal long InvalidEventCount => Interlocked.Read(ref invalidEventCount);
    internal long TruncatedEventCount => Interlocked.Read(ref truncatedEventCount);

    /// <summary>Creates a logger borrowing this runtime, including after admission closes.</summary>
    /// <exception cref="ArgumentNullException">Component is null.</exception>
    /// <exception cref="ArgumentException">Component is blank, malformed, contains controls or exceeds 255 scalars.</exception>
    public CoreLogger CreateLogger(string component)
    {
        ArgumentNullException.ThrowIfNull(component);
        if (string.IsNullOrWhiteSpace(component) || !UnicodeScalarText.IsValid(component, 255, allowControls: false))

            throw new ArgumentException("Use a component of 1-255 valid Unicode scalars without controls.", nameof(component));

    }

    internal bool IsEnabled(CoreLogLevel level)
    {
        ValidateLevel(level);
        lock (admissionGate)
        {
            return !admissionClosed && outputs.Any(output => Interested(output, level) && !output.Admission.IsFaulted);
        }
    }

    internal LogWriteResult Write(string component, CoreLogLevel level, LogEventData? data, Func<LogEventData>? factory)
    {
        ValidateLevel(level);
        bool renderConsole, renderFile;
        lock (admissionGate)
        {
            if (admissionClosed) { return Stopped(isTruncated: false); }
            if (!outputs.Any(output => Interested(output, level))) { return FinishWithoutCapture(level, LogWriteOutcome.Filtered); }
            if (!outputs.Any(output => Interested(output, level) && !output.Admission.IsFaulted))
            {
                return FinishWithoutCapture(level, LogWriteOutcome.Rejected);
            }
            renderConsole = consoleWriter is not null && outputs.Any(output => output.Name == "console" && Interested(output, level) && !output.Admission.IsFaulted);
            renderFile = fileWriter is not null && outputs.Any(output => output.Name == "file" && Interested(output, level) && !output.Admission.IsFaulted);
        }
        // No gate is held around caller factories, clock, scopes or payload callbacks.
        if (factory is not null) { data = factory(); }
        if (data is null) { return Invalid("InvalidFactoryResult"); }
        string? eventName = data.EventName;
        string? message = data.Message;
        string? template = data.MessageTemplate;
        string? operationId = data.OperationId;
        CapturedEventFacts facts;
        try
        {
            facts = EventProjection.Capture(metadata, Scopes.Capture(), component, level, data, timeProvider)!;
        }
        catch (Exception error) when (error is not (OutOfMemoryException or StackOverflowException))
        {
            return Invalid(CaptureError(eventName, message, template, operationId));
        }
        EncodedEvent captured;
        RenderedEvent? preparedConsole = null, preparedFile = null;
        // Other owned encoding/rendering consistency errors are Core defects, not invalid caller content.
        try
        {
            captured = EventEncoder.Encode(facts, maximumEntryBytes);
            if (renderConsole) { preparedConsole = consoleWriter!.Prepare(captured); }
            if (renderFile) { preparedFile = fileWriter!.Prepare(captured); }
        }
        catch (EventEnvelopeTooLargeException) { return Invalid("EnvelopeTooLarge"); }
        bool isTruncated = captured.IsTruncated || preparedConsole?.IsTruncated == true || preparedFile?.IsTruncated == true;
        lock (admissionGate)
        {
            if (isTruncated) { IncrementLossCounter(ref truncatedEventCount); }
            if (admissionClosed) { return Stopped(isTruncated); }
            return Admit(level, captured, preparedConsole?.Line, preparedFile?.Line, isTruncated);
        }
    }

    // Primitive internal qualification closure: no final flush or bounded shutdown is promised.
    // Supported applications use StopAsync/DisposeAsync; a later flush cannot reopen an exited worker.
    internal void CloseAdmission()
    {
        lock (admissionGate) { shutdownStartedAtUtc ??= DateTimeOffset.UtcNow; admissionClosed = true; queuedOutputs?.Complete(); }
    }

    private LogWriteResult Invalid(string code)
    {
        IncrementLossCounter(ref invalidEventCount);
        return new LogWriteResult(LogWriteOutcome.InvalidEvent, [], code);
    }

    // Loss counters retain a nonnegative lower bound once Int64 can no longer represent the total.
    private static void IncrementLossCounter(ref long counter)
    {
        long current = Interlocked.Read(ref counter);
        while (current != long.MaxValue)
        {
            long observed = Interlocked.CompareExchange(ref counter, current + 1, current);
            if (observed == current) { return; }
            current = observed;
        }
    }

    private LogWriteResult Stopped(bool isTruncated) => new(LogWriteOutcome.Stopped,
        outputs.Select(output => new LogOutputWriteResult(output.Name, LogOutputWriteStatus.Stopped, null)), isTruncated: isTruncated);

    private LogWriteResult FinishWithoutCapture(CoreLogLevel level, LogWriteOutcome outcome)
    {
        var observations = new List<LogOutputWriteResult>(outputs.Length);
        foreach (Output output in outputs)
        {
            LogOutputWriteStatus status;
            if (Interested(output, level))
            {
                output.Admission.RecordUnavailable();
                status = LogOutputWriteStatus.Faulted;
            }
            else
            {
                output.Admission.RecordFiltered();
                status = LogOutputWriteStatus.Filtered;
            }
            observations.Add(new LogOutputWriteResult(output.Name, status, null));
        }
        return new LogWriteResult(outcome, observations);
    }

    private LogWriteResult Admit(CoreLogLevel level, EncodedEvent captured, ReadOnlyMemory<byte>? consoleLine, ReadOnlyMemory<byte>? fileLine, bool isTruncated)
    {
        var observations = new List<LogOutputWriteResult>(outputs.Length);
        int accepted = 0;
        int interested = 0;
        foreach (Output output in outputs)
        {
            long? sequence = null;
            LogOutputWriteStatus status;
            if (!Interested(output, level))
            {
                output.Admission.RecordFiltered();
                status = LogOutputWriteStatus.Filtered;
            }
            else
            {
                interested++;
                if (output.Admission.IsFaulted)
                {
                    output.Admission.RecordUnavailable();
                    status = LogOutputWriteStatus.Faulted;
                }
                else if (output.Name == "console" && consoleLine is { } preparedLine)
                { status = queuedOutputs!.Console!.TryAdmit(captured, preparedLine, out sequence); }
                else if (output.Name == "file" && fileLine is { } preparedFileLine)
                { status = queuedOutputs!.File!.TryAdmit(captured, preparedFileLine, out sequence); }
                else { status = output.Admission.TryAdmit(captured, out sequence); }
                if (status == LogOutputWriteStatus.Accepted) { accepted++; }
            }
            if (Interested(output, level) && status is not (LogOutputWriteStatus.Accepted or LogOutputWriteStatus.QueueFull or LogOutputWriteStatus.Faulted))
            {
                throw new InvalidOperationException("Output admission returned an invalid status.");
            }
            if (status == LogOutputWriteStatus.Accepted ? sequence is null or <= 0 : sequence is not null)
            {
                throw new InvalidOperationException("Output admission returned an invalid ticket.");
            }
            observations.Add(new LogOutputWriteResult(output.Name, status, sequence));
        }
        LogWriteOutcome outcome = LogWriteOutcome.Rejected;
        if (accepted == interested) { outcome = LogWriteOutcome.Accepted; }
        else if (accepted > 0) { outcome = LogWriteOutcome.PartiallyAccepted; }
        return new LogWriteResult(outcome, observations, isTruncated: isTruncated);
    }

    private static string CaptureError(string? name, string? message, string? template, string? operationId)
    {
        foreach (string? text in new[] { name, message, template, operationId })
        {
            if (text is not null && !UnicodeScalarText.IsValid(text, int.MaxValue, allowControls: true)) { return "InvalidUnicode"; }
        }
        if (string.IsNullOrEmpty(name) || !UnicodeScalarText.IsValid(name, 255, allowControls: false)) { return "InvalidEventName"; }
        if (message is null) { return "InvalidMessage"; }
        string? operation = operationId?.Trim();
        if (!string.IsNullOrEmpty(operation) && !UnicodeScalarText.IsValid(operation, 255, allowControls: false)) { return "InvalidOperationId"; }
        // Arbitrary cooperative callback/unsupported payload failures expose only this fixed code.
        return "UnsupportedValue";
    }

    private static bool Interested(Output output, CoreLogLevel level) => level != CoreLogLevel.None && level >= output.Threshold;
    private static void ValidateSettings(CoreLogLevel console, CoreLogLevel file, int maximumBytes)
    {
        ValidateLevel(console); ValidateLevel(file);
        if (maximumBytes is < LoggingOptions.MinimumMaxEntryBytes or > LoggingOptions.MaximumMaxEntryBytes) { throw new ArgumentOutOfRangeException("options"); }
    }
    private static void ValidateLevel(CoreLogLevel level)
    {
        if (!Enum.IsDefined(level)) { throw new ArgumentOutOfRangeException(nameof(level), "Use a declared logging level."); }
    }
    private sealed record Output(string Name, CoreLogLevel Threshold, ILogOutputAdmission Admission);
}

    private static (string? TraceId, string? SpanId) CaptureActivity()
    {
        Activity? activity = Activity.Current;
        if (activity?.IdFormat != ActivityIdFormat.W3C) { return (null, null); }
        string traceId = activity.TraceId.ToHexString();

    private static bool IsValidHexId(string value, int length)
    {
        if (value.Length != length) { return false; }
        bool nonzero = false;
        foreach (char character in value)
        {
            if (!(character is >= '0' and <= '9' or >= 'a' and <= 'f')) { return false; }
            nonzero |= character != '0';
        }
        return nonzero;

        bool truncated = scope.IsTruncated || explicitAttributes.IsTruncated;

        {
            foreach ((string key, LogValue value) in map.MapProperties)
            {
                if (winners.ContainsKey(key) || winners.Count < LogValueProjection.MaximumNodes - 1) { winners[key] = value; }
                else { truncated = true; }
            }
        }
        return new LogValueProjection().Map(winners, depth: 1, truncated);
    }

                EnvironmentKind.Function => "function",
                _ => throw new InvalidOperationException("Invalid owned environment kind.")
            }),
            ["os_family"] = LogValue.String(metadata.OperatingSystem)
        };
        if (metadata.ProcessStartedAtUtc is { } started) { resource["process_started_at"] = LogValue.Timestamp(started); }

        Add("runtime_description", metadata.RuntimeDescription);

        Add("commit_id", metadata.CommitId);
        Add("build_id", metadata.BuildId);
        return LogValue.Map(resource);

        void Add(string key, string? value) { if (value is not null) { resource[key] = LogValue.String(value); } }
    }

namespace CIS.Logging.Values;

// One budget per root. Reused immutable subtrees still count once per containment occurrence.
internal sealed class LogValueProjection
{
    internal const int MaximumNodes = 1024;
    internal const int MaximumDepth = 8;
    private int remainingNodes = MaximumNodes - 1;

    internal LogValue Array(IEnumerable<LogValue> items, int depth, bool truncated = false)
    {
        var owned = new List<LogValue>();
        foreach (LogValue item in items)
        {
            if (item is null)
            {
                throw new ArgumentException("Use LogValue.Null() for explicit null array elements.", nameof(items));
            }
            LogValue? retained = Project(item, depth + 1);
            if (retained is null)
            {
                truncated = true;
                break;
            }
            owned.Add(retained);
        }
        return LogValue.OwnedArray(owned, truncated);
    }

    internal LogValue Map(IReadOnlyDictionary<string, LogValue> properties, int depth, bool truncated = false)
    {
        var owned = new Dictionary<string, LogValue>(StringComparer.Ordinal);
        foreach ((string key, LogValue value) in properties)
        {
            if (key is null || !LogValueText.IsValidKey(key) || value is null || owned.ContainsKey(key))
            {
                throw new ArgumentException("Use unique ordinal keys of 1–255 valid Unicode scalars without controls, and explicit LogValue values.", nameof(properties));
            }
            LogValue? retained = Project(value, depth + 1);
            if (retained is null)
            {
                truncated = true;
                break;
            }
            owned.Add(key, retained);
        }
        return LogValue.OwnedMap(owned, truncated);
    }

    private LogValue? Project(LogValue value, int depth)
    {
        if (remainingNodes == 0 || depth > MaximumDepth)
        {
            return null;
        }
        if (value.NodeCount <= remainingNodes && depth - 1 + value.Depth <= MaximumDepth)
        {
            remainingNodes -= value.NodeCount;
            return value;
        }
        remainingNodes--;
        return value.Kind switch
        {
            LogValueKind.Array => Array(value.ArrayItems, depth, value.IsTruncated),
            LogValueKind.Map => Map(value.MapProperties, depth, value.IsTruncated),
            _ => throw new InvalidOperationException("A supported scalar cannot exceed a positive shape budget.")
        };
    }
}

using CIS.Text;

namespace CIS.Logging.Values;

internal static class LogValueText
{
    public static bool IsValidUnicode(string value) => UnicodeScalarText.IsValid(value, int.MaxValue, allowControls: true);

    public static bool IsValidKey(string value) => value.Length > 0 && UnicodeScalarText.IsValid(value, 255, allowControls: false);

    public static bool IsInteger(string value)
    {
        int start = value.StartsWith('-') ? 1 : 0;
        if (start == value.Length)
        {
            return false;
        }
        if (value[start] == '0')
        {
            return start == 0 && value.Length == 1;
        }
        return value[start] is >= '1' and <= '9' && Digits(value.AsSpan(start + 1));
    }

    public static bool IsDecimal(string value)
    {
        int start = value.StartsWith('-') ? 1 : 0;
        int point = value.IndexOf('.');
        if (point <= start || point == value.Length - 1)
        {
            return false;
        }
        ReadOnlySpan<char> whole = value.AsSpan(start, point - start);
        bool validWhole = whole.Length == 1 && whole[0] == '0' ||
            whole[0] is >= '1' and <= '9' && Digits(whole[1..]);
        return validWhole && Digits(value.AsSpan(point + 1));
    }

    private static bool Digits(ReadOnlySpan<char> value)
    {
        foreach (char character in value)
        {
            if (character is < '0' or > '9')
            {
                return false;
            }
        }
        return true;
    }
}

    internal OutputFlushBoundary RegisterFlushBoundary()
    {
        lock (admissionGate)
        {
            long boundary = lastAcceptedSequence;
            if (!flushRequests.TryGetValue(boundary, out OutputFlushRegistration? registration))
            {
                registration = new OutputFlushRegistration(boundary);
                if (faulted || workerExiting || workerFinished || flush is null) { registration.Resolve(false); }
                else
                {
                    // Coordination is separate from bounded event slots and cannot overflow with them.
                    flushRequests.Add(boundary, registration);
                    wake.TrySetResult();
                }
            }
            return new OutputFlushBoundary(boundary, filteredCount, queueFullCount, unavailableCount, registration);
        }

    internal void StartFileFlushTimer(TimeProvider clock, TimeSpan interval)
    {
        ArgumentNullException.ThrowIfNull(clock);
        if (interval < TimeSpan.FromMilliseconds(LoggingOptions.MinimumFlushIntervalMilliseconds)
            || interval > TimeSpan.FromMilliseconds(LoggingOptions.MaximumFlushIntervalMilliseconds))
        { throw new ArgumentOutOfRangeException(nameof(interval)); }
        ITimer Create() => clock.CreateTimer(static state => ((OutputEventQueue)state!).RequestFileFlush(), this,
            Timeout.InfiniteTimeSpan, Timeout.InfiniteTimeSpan) ?? throw new InvalidOperationException("Timer creation returned null.");
        ITimer acquired;
        if (ExecutionContext.IsFlowSuppressed()) { acquired = Create(); }
        else { using (ExecutionContext.SuppressFlow()) { acquired = Create(); } }
        bool assigned = false;
        try
        {
            bool activate;
            lock (admissionGate)
            {
                if (fileFlushTimer is not null) { throw new InvalidOperationException("A file flush timer is already owned."); }
                activate = !admissionClosed && !workerExiting && !workerFinished && flush is not null;
                if (activate) { fileFlushTimer = acquired; assigned = true; }
            }
            if (!activate) { acquired.Dispose(); return; }
            if (!acquired.Change(interval, interval))
            {
                // Terminal cleanup may have detached/disposed this timer before its initial arm.
                lock (admissionGate) { if (!ReferenceEquals(fileFlushTimer, acquired)) { return; } }
                throw new InvalidOperationException("The file flush timer could not be armed.");
            }
        }
        catch
        {
            bool dispose;
            lock (admissionGate)
            {
                dispose = !assigned || ReferenceEquals(fileFlushTimer, acquired);
                if (ReferenceEquals(fileFlushTimer, acquired)) { fileFlushTimer = null; }
            }
            // An assigned/detached timer belongs to the shared disposal task; a local extra still needs cleanup.
            if (dispose) { acquired.Dispose(); }
            throw;
        }
    }
    // One cleanup operation per acquired timer. Its gate barrier also respects an outer
    // shared admission lock: custom timer code runs only after closure/fault publication.
    private Task CancelFileFlushScheduling()
    {
        lock (admissionGate)
        {
            if (fileFlushTimerDisposal is not null) { return fileFlushTimerDisposal; }
            ITimer? owned = fileFlushTimer;
            if (owned is null) { return Task.CompletedTask; }
            fileFlushTimer = null;

            {
                lock (admissionGate) { }
                await owned.DisposeAsync().ConfigureAwait(false);
            }
            Task Launch() => Task.Run(DisposeAsync);
            if (ExecutionContext.IsFlowSuppressed()) { fileFlushTimerDisposal = Launch(); }
            else { using (ExecutionContext.SuppressFlow()) { fileFlushTimerDisposal = Launch(); } }
            return fileFlushTimerDisposal;
        }
    }

    internal LogOutputDrainResult CaptureDrain(string name, OutputFlushBoundary boundary)
    {
        lock (admissionGate)
        {
            long target = boundary.AcceptedSequenceBoundary;
            // Successful writes are a prefix; terminal failure is one ticket, abandonment one tail.
            long written = Math.Min(writtenCount, target);
            long failed = failedSequence > 0 && failedSequence <= target ? 1 : 0;
            long abandoned = firstAbandonedSequence > 0 && firstAbandonedSequence <= target
                ? Math.Min(lastAbandonedSequence, target) - firstAbandonedSequence + 1 : 0;
            Task<bool> confirmation = boundary.Registration.WriterFlushCompletion;
            LogWriterFlushStatus status = !confirmation.IsCompletedSuccessfully ? LogWriterFlushStatus.Pending
                : confirmation.Result ? LogWriterFlushStatus.Succeeded : LogWriterFlushStatus.Failed;
            return new LogOutputDrainResult(name, target, written, failed, abandoned, target - written - failed - abandoned,
                status, new LogOutputRejectionSnapshot(boundary.Filtered, boundary.QueueFull, boundary.Unavailable));
        }
    }
    internal async Task WaitForCoveredRecordsAsync(string name, OutputFlushBoundary boundary, CancellationToken token)
    {
        while (true)
        {
            token.ThrowIfCancellationRequested();

            {
                if (CaptureDrain(name, boundary).Pending == 0) { return; }
                changed = (dispositionChanged ??= NewWake()).Task;
            }
            await changed.WaitAsync(token).ConfigureAwait(false);
        }

    internal LogOutputWriteStatus TryAdmit(EncodedEvent captured, ReadOnlyMemory<byte>? preparedLine, out long? acceptedSequence)
    {
        ArgumentNullException.ThrowIfNull(captured);
        lock (admissionGate)
        {
            acceptedSequence = null;
            if (admissionClosed)
            {
                if (faulted) { RecordUnavailable(); return LogOutputWriteStatus.Faulted; }
                return LogOutputWriteStatus.Stopped;
            }
            if (lastAcceptedSequence == long.MaxValue || acceptedCount == long.MaxValue)
            {
                // No wrap, negative/duplicate ticket or invented acceptance. Exhaustion is terminal.
                Fault("AdmissionExhausted"); RecordUnavailable();
                return LogOutputWriteStatus.Faulted;
            }
            long next = lastAcceptedSequence + 1;
            if (!channel.Writer.TryWrite(new AdmittedEvent(captured, next, preparedLine)))
            {
                if (queueFullCount != long.MaxValue) { queueFullCount++; }
                return LogOutputWriteStatus.QueueFull;
            }
            lastAcceptedSequence = next;
            acceptedCount++;
            wake.TrySetResult();
            acceptedSequence = next;
            return LogOutputWriteStatus.Accepted;
        }
    }
    public void RecordFiltered() { lock (admissionGate) { if (filteredCount != long.MaxValue) { filteredCount++; } } }
    public void RecordUnavailable() { lock (admissionGate) { if (unavailableCount != long.MaxValue) { unavailableCount++; } } }
    // Primitive queue closure, not the public runtime's flush/deadline/shutdown protocol.
    internal void Complete()
    {
        lock (admissionGate)
        {
            stopRequested = true; admissionClosed = true;
            state = workerFinished ? LogOutputState.Stopped : LogOutputState.Stopping;
            channel.Writer.TryComplete();
            wake.TrySetResult();
            _ = CancelFileFlushScheduling();
        }
    }
    // Called under the shared runtime gate before the immutable deadline observation.
    internal void AbandonUnstarted()
    {
        lock (admissionGate)
        {
            stopDeadlineExpired = true;
            while (channel.Reader.TryRead(out AdmittedEvent abandoned))
            {
                if (firstAbandonedSequence == 0) { firstAbandonedSequence = abandoned.AcceptedSequence; }
                lastAbandonedSequence = abandoned.AcceptedSequence; abandonedCount++;
            }
            // A boundary covering abandoned tickets can never reach writtenCount.
            // Preserve already-started flush I/O until its actual completion.
            foreach (long target in flushRequests.Keys.ToArray())
            {
                OutputFlushRegistration registration = flushRequests[target];
                if (firstAbandonedSequence > 0 && target >= firstAbandonedSequence && !ReferenceEquals(registration, flushing))
                { registration.Resolve(false); flushRequests.Remove(target); }
            }
            NotifyDispositionChanged();
            wake.TrySetResult();
        }
    }
    internal void RequestIoCancellation()
    {
        if (ioCancellation is null) { return; }
        TaskCompletionSource completion;
        lock (admissionGate)
        {
            if (ioCancellationClosed || ioCancellationCompletion is not null) { return; }
            completion = NewWake(); ioCancellationCompletion = completion.Task;
        }
        // Stream cancellation callbacks may block or throw. They run asynchronously,
        // outside admission synchronization, and cannot extend the stop deadline.
        _ = CancelAsync();
        async Task CancelAsync()
        {
            try { await ioCancellation.CancelAsync().ConfigureAwait(false); completion.TrySetResult(); }
            catch (Exception error) { completion.TrySetException(error); }
        }
    }
    private async Task CloseIoCancellationAsync()
    {
        Task? pending;
        lock (admissionGate) { ioCancellationClosed = true; pending = ioCancellationCompletion; }
        try { if (pending is not null) { await pending.ConfigureAwait(false); } }
        finally { ioCancellation?.Dispose(); }
    }
    // Caller holds the shared gate. Publish closure, safe metadata and all unstarted loss together.
    private void Fault(string code)
    {
        faulted = true; admissionClosed = true;
        _ = CancelFileFlushScheduling();
        state = stopRequested ? LogOutputState.Stopping : LogOutputState.Faulted;
        lastFaultCode = code; lastFaultAtUtc = DateTimeOffset.UtcNow;
        channel.Writer.TryComplete();
        while (channel.Reader.TryRead(out AdmittedEvent abandoned))
        {
            if (firstAbandonedSequence == 0) { firstAbandonedSequence = abandoned.AcceptedSequence; }
            lastAbandonedSequence = abandoned.AcceptedSequence; abandonedCount++;
        }
        NotifyDispositionChanged();
        foreach (OutputFlushRegistration registration in flushRequests.Values)
        {
            if (!ReferenceEquals(registration, flushing)) { registration.Resolve(false); }
        }
        OutputFlushRegistration? active = flushing;
        flushRequests.Clear();
        // An already-started flush remains unresolved until its actual I/O settles, even if admission faults.
        if (active is not null) { flushRequests.Add(active.AcceptedSequenceBoundary, active); }
        wake.TrySetResult();
    }
    // Called only after queue closure or worker exit; cleanup does not reclassify delivered records or restart output work.
    internal void RecordCleanupFault(string code)
    {
        lock (admissionGate)
        {
            faulted = true; admissionClosed = true;
            lastFaultCode = code; lastFaultAtUtc = DateTimeOffset.UtcNow;
            state = workerFinished && stopRequested ? LogOutputState.Stopped : stopRequested ? LogOutputState.Stopping : LogOutputState.Faulted;
            foreach (OutputFlushRegistration registration in flushRequests.Values)
            {
                if (!ReferenceEquals(registration, flushing)) { registration.Resolve(false); }
            }
            flushRequests.Clear();
            // Actual flush I/O keeps its own registration until it settles.
            if (flushing is not null) { flushRequests.Add(flushing.AcceptedSequenceBoundary, flushing); }
            wake.TrySetResult();
        }
    }
    private async Task ProcessAsync()
    {
        Exception? primaryFailure = null;
        try
        {
            while (true)
            {
                AdmittedEvent admitted = default;
                OutputFlushRegistration? request = null;
                Task? wait = null;
                lock (admissionGate)
                {
                    if (faulted) { workerExiting = true; return; }
                    if (stopDeadlineExpired)
                    {
                        // Expiry cancels active I/O; it never starts a later flush.
                        foreach (OutputFlushRegistration registration in flushRequests.Values) { registration.Resolve(false); }
                        flushRequests.Clear();
                    }
                    if (flushRequests.Count > 0 && flushRequests.First().Key <= writtenCount)
                    {
                        request = flushRequests.First().Value;
                        flushing = request;
                    }
                    else if (channel.Reader.TryRead(out admitted)) { inFlight = true; inFlightSequence = admitted.AcceptedSequence; }
                    else if (stopRequested) { workerExiting = true; return; }
                    else
                    {
                        // Reset only while no work is visible under the admission gate: no lost wakeup.
                        if (wake.Task.IsCompleted) { wake = NewWake(); }
                        wait = wake.Task;
                    }
                }
                if (wait is not null) { await wait.ConfigureAwait(false); continue; }
                if (request is not null)
                {
                    // Flush at this fixed boundary before starting a later record. All I/O is outside the gate.
                    await flush!(ioCancellation?.Token ?? CancellationToken.None).ConfigureAwait(false);
                    lock (admissionGate)
                    {
                        flushRequests.Remove(request.AcceptedSequenceBoundary);
                        flushing = null;
                        request.Resolve(true);
                    }
                }
                else
                {
                    // Complete record confirmation only; a dequeued/partially emitted write is still pending.
                    await write(admitted, ioCancellation?.Token ?? CancellationToken.None).ConfigureAwait(false);
                    lock (admissionGate) { writtenCount++; inFlight = false; NotifyDispositionChanged(); }
                }
            }
        }
        catch (Exception error)
        {
            primaryFailure = error;
            lock (admissionGate)
            {
                if (inFlight) { failedCount++; failedSequence = inFlightSequence; inFlight = false; }
                bool flushFailed = flushing is not null;
                flushing?.Resolve(false);
                flushing = null;
                Fault(flushFailed ? "FlushFailed" : error is FileOutputWriteException fileError ? fileError.FaultCode : "WriteFailed");
            }
            throw;
        }
        finally
        {
            // No pending writer call remains. Owned cleanup precedes terminal state/Completion,
            // outside the gate, and never receives a borrowed console stream.
            lock (admissionGate) { workerExiting = true; }
            List<Exception>? cleanupFailures = null;
            try { await CloseIoCancellationAsync().ConfigureAwait(false); }
            catch (Exception error) { RecordCleanupFault("CancellationCallbacksFailed"); (cleanupFailures ??= []).Add(error); }
            try { await CancelFileFlushScheduling().ConfigureAwait(false); }
            catch (Exception error) { RecordCleanupFault("FileFlushTimerCleanupFailed"); (cleanupFailures ??= []).Add(error); }
            try { ownedResource?.Dispose(); }
            catch (Exception error) { RecordCleanupFault("OutputCleanupFailed"); (cleanupFailures ??= []).Add(error); }
            lock (admissionGate)
            {
                admissionClosed = true; channel.Writer.TryComplete();
                workerExiting = true; workerFinished = true;
                state = stopRequested || !faulted ? LogOutputState.Stopped : LogOutputState.Faulted;
            }
            if (cleanupFailures is not null)
            {
                // Preserve the exact original I/O failure as well as every failed cleanup step.
                if (primaryFailure is not null) { cleanupFailures.Insert(0, primaryFailure); }
                if (cleanupFailures.Count == 1) { ExceptionDispatchInfo.Capture(cleanupFailures[0]).Throw(); }
                throw new AggregateException("Output cleanup failed.", cleanupFailures);
            }
        }
    }
}

namespace CIS.Logging.Runtime;

// Creates exactly one independent worker for each enabled destination and no disabled worker.
internal sealed class OutputQueues
{
    private readonly object admissionGate;
    internal OutputQueues(int capacity, bool consoleEnabled, bool fileEnabled,
        Func<AdmittedEvent, ValueTask>? consoleWrite, Func<AdmittedEvent, ValueTask>? fileWrite, object? sharedAdmissionGate = null, Func<Func<Task>, Task>? launchWorker = null, IDisposable? fileResource = null, Func<ValueTask>? consoleFlush = null, Func<ValueTask>? fileFlush = null,
        Func<AdmittedEvent, CancellationToken, ValueTask>? consoleCancellableWrite = null, Func<AdmittedEvent, CancellationToken, ValueTask>? fileCancellableWrite = null,
        Func<CancellationToken, ValueTask>? consoleCancellableFlush = null, Func<CancellationToken, ValueTask>? fileCancellableFlush = null)
    {
        admissionGate = sharedAdmissionGate ?? new object();
        if (capacity is < LoggingOptions.MinimumQueueCapacity or > LoggingOptions.MaximumQueueCapacity) { throw new ArgumentOutOfRangeException(nameof(capacity)); }
        if (consoleEnabled != (consoleWrite is not null)) { throw new ArgumentException("Writer presence must match enabled console output.", nameof(consoleWrite)); }
        if (fileEnabled != (fileWrite is not null)) { throw new ArgumentException("Writer presence must match enabled file output.", nameof(fileWrite)); }
        if (!consoleEnabled && consoleFlush is not null) { throw new ArgumentException("Disabled console output cannot have a flush callback.", nameof(consoleFlush)); }
        if (!fileEnabled && fileFlush is not null) { throw new ArgumentException("Disabled file output cannot have a flush callback.", nameof(fileFlush)); }
        try
        {
            if (consoleWrite is not null) { Console = Create("console", consoleWrite, null, consoleFlush, consoleCancellableWrite, consoleCancellableFlush); }
            if (fileWrite is not null) { File = Create("file", fileWrite, fileResource, fileFlush, fileCancellableWrite, fileCancellableFlush); }
            Completion = Task.WhenAll(Console?.Completion ?? Task.CompletedTask, File?.Completion ?? Task.CompletedTask);
        }
        catch
        {
            // Unpublished queues are empty. Close/join only this attempt's idle workers, outside the gate.
            Complete();
            try { Task.WhenAll(Console?.Completion ?? Task.CompletedTask, File?.Completion ?? Task.CompletedTask).GetAwaiter().GetResult(); } catch { }
            throw;
        }
        OutputEventQueue Create(string name, Func<AdmittedEvent, ValueTask> write, IDisposable? ownedResource, Func<ValueTask>? flush, Func<AdmittedEvent, CancellationToken, ValueTask>? cancellableWrite, Func<CancellationToken, ValueTask>? cancellableFlush)
        {
            try { return new OutputEventQueue(capacity, write, admissionGate, launchWorker, ownedResource, flush, cancellableWrite, cancellableFlush); }
            catch { throw new CoreLogInitializationException(name, "WorkerInitializationFailed"); }
        }
    }
    internal OutputEventQueue? Console { get; }
    internal OutputEventQueue? File { get; }
    internal Task Completion { get; }
    internal void Complete() { lock (admissionGate) { Console?.Complete(); File?.Complete(); } }
    internal void AbandonUnstarted() { lock (admissionGate) { Console?.AbandonUnstarted(); File?.AbandonUnstarted(); } }
