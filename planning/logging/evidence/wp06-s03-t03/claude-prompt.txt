    [ValidateSet('Coverage', 'Mutation')][string] $Mode = 'Coverage',
    [string] $OutputRoot

$ErrorActionPreference = 'Stop'
Set-StrictMode -Version Latest
$languageRoot = Split-Path $PSScriptRoot -Parent
$repositoryRoot = Split-Path (Split-Path $languageRoot -Parent) -Parent
$policy = Get-Content -Raw -LiteralPath (Join-Path $PSScriptRoot 'toolchain.json') | ConvertFrom-Json
if (-not $OutputRoot) { $OutputRoot = Join-Path $repositoryRoot ('.artifacts/csharp/microsoft-logging/' + [guid]::NewGuid().ToString('N')) }
$OutputRoot = [IO.Path]::GetFullPath($OutputRoot)
if (Test-Path -LiteralPath $OutputRoot) { throw 'Use a new output directory; retained reports are immutable.' }
New-Item -ItemType Directory -Path (Join-Path $OutputRoot 'logs') -Force | Out-Null
$commands = [Collections.Generic.List[object]]::new()
$sourceInputs = @(Get-ChildItem -LiteralPath (Join-Path $languageRoot 'src/CIS.Core/Logging/MicrosoftLogger') -File -Filter '*.cs' | ForEach-Object {
    @{ path = [IO.Path]::GetRelativePath($repositoryRoot, $_.FullName).Replace('\', '/'); sha256 = (Get-FileHash -LiteralPath $_.FullName -Algorithm SHA256).Hash.ToLowerInvariant() }
})
$sourceInputs | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $OutputRoot 'adapter-source-inputs.json') -Encoding utf8
function Invoke-Recorded {
    param([string] $Name, [string[]] $Arguments)
    $log = Join-Path $OutputRoot "logs/$Name.log"
    & dotnet @Arguments 2>&1 | Tee-Object -FilePath $log | Out-Host
    $code = $LASTEXITCODE
    $commands.Add(@{ name = $Name; executable = 'dotnet'; argv = $Arguments; workingDirectory = (Get-Location).Path; exitCode = $code; log = "logs/$Name.log" })
    $commands | ConvertTo-Json -Depth 7 | Set-Content -LiteralPath (Join-Path $OutputRoot 'commands.json') -Encoding utf8
    if ($code -ne 0) { throw "$Name failed with exit $code. Native outputs retained." }
}
Push-Location $languageRoot
try {
    if ((& dotnet --version).Trim() -ne $policy.sdk) { throw 'Selected SDK differs from toolchain policy.' }
    if ($Mode -eq 'Coverage') {
        foreach ($family in @('Unit', 'Integration')) {
            $project = "tests/CIS.Core.$($family)Tests/CIS.Core.$($family)Tests.csproj"
            Invoke-Recorded "build-$family" @('build', $project, '-c', 'Release', '--no-restore')
            $directory = Join-Path $OutputRoot $family.ToLowerInvariant()
            Invoke-Recorded "coverage-$family" @('test', $project, '-c', 'Release', '--no-build', '--no-restore', '--filter', "FullyQualifiedName~CIS.Tests.$($family)Tests.Provider", '--settings', 'verification/coverage/microsoft-logging.runsettings', '--collect', 'Code Coverage;Format=cobertura', '--results-directory', $directory, '--logger', 'trx;LogFileName=report.trx')
        }
    }
    else {
        # Stryker must never write instrumented sources/assemblies into the shared checkout.
        $snapshot = Join-Path $OutputRoot 'source'
        Push-Location $repositoryRoot
        try {
            $files = @(& git ls-files --cached --others --exclude-standard -- implementations/csharp)
            if ($LASTEXITCODE -ne 0) { throw 'Cannot enumerate maintained mutation inputs.' }
            $files += 'README.md'
            $inventory = foreach ($relative in ($files | Sort-Object -Unique)) {
                $source = Join-Path $repositoryRoot $relative
                $copy = Join-Path $snapshot $relative
                New-Item -ItemType Directory -Path (Split-Path $copy -Parent) -Force | Out-Null
                Copy-Item -LiteralPath $source -Destination $copy
                @{ path = $relative; sha256 = (Get-FileHash -LiteralPath $copy -Algorithm SHA256).Hash.ToLowerInvariant() }
            }
            $inventory | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $OutputRoot 'snapshot-inputs.json') -Encoding utf8
        }
        finally { Pop-Location }
        $isolatedLanguage = Join-Path $snapshot 'implementations/csharp'
        Set-Location $isolatedLanguage
        Invoke-Recorded 'tool-restore' @('tool', 'restore', '--configfile', 'eng/Tools.NuGet.config')
        Invoke-Recorded 'restore-unit' @('restore', 'tests/CIS.Core.UnitTests/CIS.Core.UnitTests.csproj', '--locked-mode', '--configfile', 'NuGet.config')
        Invoke-Recorded 'build-unit' @('build', 'tests/CIS.Core.UnitTests/CIS.Core.UnitTests.csproj', '-c', 'Release', '--no-restore')
        Invoke-Recorded 'baseline-unit' @('test', 'tests/CIS.Core.UnitTests/CIS.Core.UnitTests.csproj', '-c', 'Release', '--no-build', '--no-restore', '--filter', 'FullyQualifiedName~CIS.Tests.UnitTests.Provider', '--logger', 'trx;LogFileName=baseline.trx', '--results-directory', (Join-Path $OutputRoot 'baseline'))
        $packageDirectory = ((& dotnet nuget locals global-packages --list) -join '').Split(': ', 2)[1].Trim()
        $tool = Join-Path $packageDirectory 'dotnet-stryker/4.16.0/tools/net8.0/any/Stryker.CLI.dll'
        if (-not (Test-Path -LiteralPath $tool)) { throw 'Pinned Stryker entry point was not materialized.' }
        Set-Location (Join-Path $isolatedLanguage 'src/CIS.Core')
        $mutationFailure = $null
        try {
            Invoke-Recorded 'mutation' @('exec', '--fx-version', $policy.toolHostRuntime, '--roll-forward', 'Disable', $tool, '--config-file', '../../verification/mutation/microsoft-logging-config.json', '--output', (Join-Path $OutputRoot 'mutation'))
        }
        catch { $mutationFailure = $_ }
        finally {
            foreach ($input in $inventory) {

            }
            Set-Location $isolatedLanguage
            Invoke-Recorded 'unmutated-rebuild' @('build', 'tests/CIS.Core.UnitTests/CIS.Core.UnitTests.csproj', '-c', 'Release', '--no-restore', '--no-incremental')
        }
        if ($mutationFailure) { throw $mutationFailure }
    }
    Write-Output "Microsoft logging $Mode completed. Native reports: $OutputRoot"
}
finally {
    try {
        foreach ($input in $sourceInputs) {
            if ((Get-FileHash -LiteralPath (Join-Path $repositoryRoot $input.path) -Algorithm SHA256).Hash.ToLowerInvariant() -ne $input.sha256) { throw 'Canonical adapter source changed during verification.' }
        }
    }
    finally { Pop-Location }
}

# Mutation verification

Run from the repository root:

```powershell
pwsh -File implementations/csharp/eng/verify-toolchain.ps1
```

The driver copies maintained source into a unique `.artifacts/csharp/toolchain/<run>/source` directory and uses fresh CLI and NuGet caches. Stryker **4.16.0** runs on the explicitly selected .NET **8.0.31** tool host with roll-forward disabled, mutates the genuine Core severity parser with the `Complete` mutation level, and runs its MSTest cases through **VSTest** against `net10.0`. Local JSON/HTML reporters are enabled; no dashboard reporter is selected.

[stryker-config.json](stryker-config.json) is the retained smoke scope, not a future full-release mutation matrix. The parser run remains a smoke scope. Core now contains logging/provider and other module implementations; their full shipping scope and materially different targets must be covered by the WP07 release matrix.

The driver rejects zero valid mutants and checks `Killed / (Killed + Timeout + Survived + NoCoverage)` at 90% or above. Every testable mutant must be killed or explicitly diagnosed; native noncompiling/ignored mutants require a source/tool-bound review in `smoke-dispositions.json`. A changed source hash, tool pin or unexpected exclusion fails qualification. No excluded mutant receives kill credit. Native reports remain immutable. Source hashes must match before/after mutation, followed by a fresh unmutated build, pack and shipping dependency check in the isolated copy.

Read [toolchain qualification](../../../../docs/architecture/csharp-toolchain.md) for reports, prerequisites and retention limits. Retain the complete local run directory; generated caches and packages are ignored by Git.

using Microsoft.Extensions.Logging;

namespace CIS.Logging.MicrosoftLogger;

/// <summary>Creates Microsoft category loggers that borrow an application-owned core runtime.</summary>
/// <remarks>Disposal disables this provider's future calls and never stops or disposes the shared runtime.
/// A call already running when disposal occurs may finish. Category names follow the core component rules.
/// Supplied rendered messages and exceptions enter core capture. Null or empty EventId.Name becomes
/// log.message; EventId.Id supplies the integer attribute event_id, including zero and negative values.
/// Structured state preserves {OriginalFormat} without parsing. Core checks interest before invoking
/// formatter or copying state; full queues remain interested. Formatter exceptions propagate unchanged;
/// invalid retained payload increments core diagnostics without output admission. Oversized valid payload
/// is truncated. Disposed, stopped, filtered and all-faulted calls skip both callbacks after argument guards.
/// Healthy outputs admit independently. Provider-created scopes are copied at BeginScope; borrowed host
/// scopes are copied during accepted capture, after Activity and clock snapshots. Combined scope depth is 32.</remarks>
public sealed class CoreLoggerProvider : ILoggerProvider, ISupportExternalScope
{
    private readonly CoreLogRuntime runtime;
    private int disposed;
    private readonly object scopeProviderGate = new();
    private IExternalScopeProvider? externalScopes;

    /// <summary>Creates a nonowning provider over an existing runtime.</summary>
    /// <exception cref="ArgumentNullException">Runtime is null.</exception>
    public CoreLoggerProvider(CoreLogRuntime runtime)
    {
        ArgumentNullException.ThrowIfNull(runtime);
        this.runtime = runtime;
    }

    /// <summary>Creates a category logger using the core's validated component name.</summary>
    /// <exception cref="ObjectDisposedException">This provider has been disposed.</exception>
    /// <exception cref="ArgumentNullException">Category name is null.</exception>
    /// <exception cref="ArgumentException">The name does not satisfy the core component rules.</exception>
    public ILogger CreateLogger(string categoryName)
    {
        ObjectDisposedException.ThrowIf(IsDisposed, this);
        return new CategoryLogger(this, runtime.CreateLogger(categoryName));
    }

    private bool IsDisposed => Volatile.Read(ref disposed) != 0;

    /// <summary>Idempotently disables this provider; the application retains runtime ownership.</summary>
    public void Dispose()
    {
        lock (scopeProviderGate) { Interlocked.Exchange(ref disposed, 1); Volatile.Write(ref externalScopes, null); }
    }

    /// <summary>Borrows the host scope source; accepted events copy its active scopes inline.</summary>
    /// <remarks>Existing core/provider scopes are the outer layer, host scopes the inner layer, and event attributes replace matching retained keys.
    /// Host scope lifetime belongs to the host. Provider disposal releases this reference and never disposes host handles.</remarks>
    /// <exception cref="ArgumentNullException">Scope provider is null.</exception>
    /// <exception cref="ObjectDisposedException">This provider is disposed.</exception>
    public void SetScopeProvider(IExternalScopeProvider scopeProvider)
    {
        ArgumentNullException.ThrowIfNull(scopeProvider);
        lock (scopeProviderGate) { ObjectDisposedException.ThrowIf(IsDisposed, this); Volatile.Write(ref externalScopes, scopeProvider); }
    }

    private static CoreLogLevel Level(LogLevel logLevel)
    {
        if (!Enum.IsDefined(logLevel)) { throw new ArgumentOutOfRangeException(nameof(logLevel)); }
        // The selected v1 Core and Microsoft severity numbers are identical (Trace through None).
        return (CoreLogLevel)logLevel;
    }

    private sealed class CategoryLogger(CoreLoggerProvider provider, CoreLogger core) : ILogger
    {
        public bool IsEnabled(LogLevel logLevel)
        {
            CoreLogLevel level = Level(logLevel);
            return !provider.IsDisposed && core.IsEnabled(level);
        }

        public IDisposable? BeginScope<TState>(TState state) where TState : notnull
        {
            ArgumentNullException.ThrowIfNull(state);
            ObjectDisposedException.ThrowIf(provider.IsDisposed, provider);
            if (provider.runtime.IsAdmissionClosed) { throw new InvalidOperationException("Cannot begin a scope after runtime admission closes."); }
            return provider.runtime.Scopes.BeginCopiedScope(() => ProviderScopeCapture.Copy(state));
        }

        public void Log<TState>(LogLevel logLevel, EventId eventId, TState state, Exception? exception,
            Func<TState, Exception?, string> formatter)
        {
            ArgumentNullException.ThrowIfNull(formatter);
            CoreLogLevel level = Level(logLevel);
            if (provider.IsDisposed) { return; }
            IExternalScopeProvider? scopes = Volatile.Read(ref provider.externalScopes);
            _ = core.WriteLazy(level, () => new LogEventData
            {
                EventName = string.IsNullOrEmpty(eventId.Name) ? "log.message" : eventId.Name,
                Message = formatter(state, exception),
                Exception = exception,
                CaptureState = () => ProviderStateCapture.Capture(state, eventId.Id),
                CaptureScopes = scopes is null ? null : depth => ProviderScopeCapture.Capture(scopes, depth)
            });
        }
    }
}

using CIS.Logging.Scopes;
using Microsoft.Extensions.Logging;

namespace CIS.Logging.MicrosoftLogger;

// Host scopes are borrowed; only bounded immutable Core payloads leave this callback.
internal static class ProviderScopeCapture
{
    internal static CapturedOperationContext Copy(object? state) => state switch
    {
        OperationContextData typed => CapturedOperationContext.Capture(typed),
        IEnumerable<KeyValuePair<string, object?>> fields => ProviderStateCapture.CaptureScope(fields),
        _ => throw new ArgumentException("Use OperationContextData or structured string/object fields for scopes.", nameof(state))
    };

    internal static CapturedScopeContext Capture(IExternalScopeProvider scopes, int maximumDepth)
    {
        OperationScopeFrame? head = null; int depth = 0;
        scopes.ForEachScope((state, _) =>
        {
            if (++depth > maximumDepth) { throw new InvalidOperationException("The maximum combined scope depth is 32."); }
            CapturedOperationContext context;
            try { context = Copy(state); }
            catch (ArgumentException) { context = CapturedOperationContext.Omitted; }
            head = new OperationScopeFrame(context, head);
        }, 0);
        return CapturedScopeContext.FromHead(head);
    }
}

// A single bounded synchronous copy; no template parsing or arbitrary object serialization.
internal sealed class ProviderStateCapture
{

    private ProviderStateCapture(bool includeEventId) => remaining = LogValueProjection.MaximumNodes - (includeEventId ? 2 : 1);
    private readonly HashSet<object> ancestors = new(ReferenceEqualityComparer.Instance);

    internal static (string? Template, LogValue Attributes) Capture<TState>(TState state, int eventId)
    {
        var captured = CaptureRoot(state, eventId, includeEventId: true);
        return (captured.Template, captured.Attributes);
    }

    internal static CapturedOperationContext CaptureScope(IEnumerable<KeyValuePair<string, object?>> state)
    {

    private static (string? Template, LogValue Attributes, string? OperationId) CaptureRoot<TState>(TState state, int eventId, bool includeEventId)
    {

        var attributes = new Dictionary<string, LogValue>(StringComparer.Ordinal);
        if (includeEventId) { attributes["event_id"] = LogValue.Integer(eventId); }
        if (state is not IEnumerable<KeyValuePair<string, object?>> fields)
        { return (null, LogValue.OwnedMap(attributes, truncated: false), null); }
        string? template = null, operationId = null; bool truncated = false; int visited = 0;
        capture.Enter(fields);
        try
        {
            foreach ((string key, object? value) in fields)
            {
                if (++visited > LogValueProjection.MaximumNodes) { truncated = true; break; }
                if (key == "{OriginalFormat}")
                {
                    if (value is not null and not string) { throw new ArgumentException("OriginalFormat must be a string or null."); }
                    template = (string?)value; continue;

                if (!includeEventId && key == "operation_id")
                {
                    if (value is string identifier)
                    {
                        operationId = CapturedOperationContext.NormalizeOperationId(identifier, "state");
                        attributes.Remove(key); continue;
                    }
                    operationId = null;
                }
                if (includeEventId && key == "event_id") { continue; }
                // Keep scanning bounded root metadata after the value budget is spent.
                if (capture.remaining == 0) { truncated = true; continue; }
                ValidateKey(key);
                LogValue? retained = capture.Value(value, depth: 2);
                if (retained is null) { truncated = true; continue; }
                attributes[key] = retained;
            }
        }
        finally { capture.ancestors.Remove(fields); }
        return (template, LogValue.OwnedMap(attributes, truncated), operationId);

            float scalar => LogValue.Number(scalar),
            double scalar => LogValue.Number(scalar),
            decimal scalar => Decimal(scalar),
            Guid scalar => LogValue.Uuid(scalar),
            DateTimeOffset scalar => LogValue.Timestamp(scalar),
            DateTime scalar when scalar.Kind == DateTimeKind.Utc => LogValue.Timestamp(new DateTimeOffset(scalar)),
            IEnumerable<KeyValuePair<string, object?>> properties => Map(properties, depth),
            IDictionary properties => Map(Entries(properties), depth, owner: properties),
            IEnumerable items => Array(items, depth),
            _ => throw new ArgumentException("Use supported scalar, collection or explicit LogValue inputs.")
        };
    }

    private LogValue Array(IEnumerable items, int depth, bool truncated = false)
    {
        Enter(items);
        try
        {
            var owned = new List<LogValue>();
            foreach (object? item in items)
            {
                LogValue? retained = Value(item, depth + 1);
                if (retained is null) { truncated = true; break; }
                owned.Add(retained);
            }
            return LogValue.OwnedArray(owned, truncated);
        }
        finally { ancestors.Remove(items); }
    }

    private LogValue Map(IEnumerable<KeyValuePair<string, object?>> properties, int depth, bool truncated = false, object? owner = null)
    {
        owner ??= properties; Enter(owner);
        try
        {
            var owned = new Dictionary<string, LogValue>(StringComparer.Ordinal);
            foreach ((string key, object? value) in properties)
            {
                ValidateKey(key);
                LogValue? retained = Value(value, depth + 1);
                if (retained is null) { truncated = true; break; }
                owned[key] = retained;
            }
            return LogValue.OwnedMap(owned, truncated);
        }
        finally { ancestors.Remove(owner); }
    }

    private static IEnumerable<KeyValuePair<string, object?>> Entries(IDictionary properties)
    {
        foreach (DictionaryEntry entry in properties)
        { yield return new(entry.Key as string ?? throw new ArgumentException("Map keys must be strings."), entry.Value); }
    }

    private void Enter(object collection)
    {
        if (!ancestors.Add(collection)) { throw new ArgumentException("Cyclic state is unsupported."); }
    }
    private static void ValidateKey(string key)
    {
        if (key is null || !LogValueText.IsValidKey(key)) { throw new ArgumentException("Use valid ordinal attribute keys."); }
    }
    private static LogValue Decimal(decimal value)
    {
        string text = value.ToString(CultureInfo.InvariantCulture);
        return LogValue.DecimalString(text.Contains('.', StringComparison.Ordinal) ? text : text + ".0");
    }
}
